@OptIn(ExperimentalLayoutApi::class)
@Composable
fun StrategyScreen(
    currentConfig: StrategyConfig?,
    customStrategies: List<CustomPatternStrategy> = emptyList(),
    patternOccurrences: List<PatternOccurrenceRecord> = emptyList(),
    onSaveCustomStrategy: (CustomPatternStrategy) -> Unit = {},
    onToggleCustomStrategyEnabled: (CustomPatternStrategy, Boolean) -> Unit = { _, _ -> },
    onToggleCustomStrategyNotifications: (CustomPatternStrategy, Boolean) -> Unit = { _, _ -> },
    onDeleteCustomStrategy: (Long) -> Unit = {},
    onReorderStrategies: (CustomPatternStrategy, CustomPatternStrategy) -> Unit = { _, _ -> },
    onSaveConfig: (StrategyConfig) -> Unit
) {
    val context = LocalContext.current
    val config = currentConfig ?: StrategyConfig()

    var lowThreshText by remember(config.lowThreshold) {
        mutableStateOf(String.format(Locale.US, "%.2f", config.lowThreshold))
    }
    var bigThreshText by remember(config.bigThreshold) {
        mutableStateOf(String.format(Locale.US, "%.2f", config.bigThreshold))
    }
    var defaultConsecutiveText by remember(config.strategy1RequiredLowRounds) {
        mutableStateOf(config.strategy1RequiredLowRounds.toString())
    }
    var monitoringEnabled by remember(config.monitoringEnabled) {
        mutableStateOf(config.monitoringEnabled)
    }
    var maxAttemptsText by remember(config.maxStrategyBetAttempts) {
        mutableStateOf(config.maxStrategyBetAttempts.toString())
    }
    var martingaleEnabled by remember(config.isMartingaleEnabled) {
        mutableStateOf(config.isMartingaleEnabled)
    }
    var lossProgressionEnabled by remember(config.lossProgressionEnabled) {
        mutableStateOf(config.lossProgressionEnabled)
    }
    var lossProgressionMode by remember(config.lossProgressionMode) {
        mutableStateOf(config.lossProgressionMode)
    }
    var lossProgressionAddAmountText by remember(config.lossProgressionAddAmount) {
        mutableStateOf(String.format(Locale.US, "%.2f", config.lossProgressionAddAmount))
    }

    var selectedGroupFilter by remember { mutableStateOf<StrategyGroup?>(null) }
    var editingCustomStrategy by remember { mutableStateOf<CustomPatternStrategy?>(null) }
    var preselectedGroup by remember { mutableStateOf(StrategyGroup.MULTIPLIER_INTERVAL) }
    var preselectedMode by remember { mutableStateOf(StrategyMode.STRATEGY_A_CONSECUTIVE_LOWS) }
    var preselectedSubtype by remember { mutableStateOf(MultiplierRuleSubtype.STANDARD_THRESHOLD) }
    var showCustomBuilder by remember { mutableStateOf(false) }

    val orderedStrategies = remember(customStrategies, selectedGroupFilter) {
        val sorted = customStrategies.sortedWith(compareBy<CustomPatternStrategy> { it.sortOrder }.thenBy { it.id })
        if (selectedGroupFilter == null) sorted else sorted.filter { it.effectiveGroup() == selectedGroupFilter }
    }

    Box(modifier = Modifier.fillMaxSize()) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .background(MaterialTheme.colorScheme.background)
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
        Spacer(modifier = Modifier.height(8.dp))
        Text(
            text = "STRATEGY MANAGER",
            style = MaterialTheme.typography.titleMedium,
            fontWeight = FontWeight.Bold,
            color = PrimaryCobalt
        )
        Text(
            text = "3 Strategy Groups: (A) Multiplier Interval (LOW / ABOVE->BELOW / BIG), (B) Time Interval (2–55m wait & 1h/3h/6h/Custom frequency), and (C) Color Sequence (Horizontal & Vertical).",
            fontSize = 11.sp,
            color = MaterialTheme.colorScheme.onSurfaceVariant
        )

        // Strategy Group Organizer Tabs
        FlowRow(
            horizontalArrangement = Arrangement.spacedBy(6.dp),
            verticalArrangement = Arrangement.spacedBy(4.dp)
        ) {
            FilterChip(
                selected = selectedGroupFilter == null,
                onClick = { selectedGroupFilter = null },
                label = { Text("All Groups (${customStrategies.size})", fontSize = 11.sp, fontWeight = FontWeight.Bold) },
                modifier = Modifier.testTag("strategy_group_tab_ALL")
            )
            StrategyGroup.entries.forEach { group ->
                val count = customStrategies.count { it.effectiveGroup() == group }
                FilterChip(
                    selected = selectedGroupFilter == group,
                    onClick = { selectedGroupFilter = group },
                    label = { Text("${group.displayTitle} ($count)", fontSize = 11.sp, fontWeight = FontWeight.Bold) },
                    modifier = Modifier.testTag("strategy_group_tab_${group.name}")
                )
            }
        }

        // Active & Saved Strategies List + Reordering + Per-Strategy Occurrence History
        Card(
            modifier = Modifier
                .fillMaxWidth()
                .testTag("custom_pattern_strategies_card"),
            shape = RoundedCornerShape(16.dp),
            colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface),
            border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant)
        ) {
            Column(
                modifier = Modifier.padding(12.dp),
                verticalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Text(
                    text = "USER STRATEGIES & INDEPENDENT HISTORIES (${orderedStrategies.size})",
                    fontWeight = FontWeight.Bold,
                    fontSize = 12.sp,
                    color = PrimaryCobalt
                )

                if (orderedStrategies.isEmpty()) {
                    Surface(
                        color = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.5f),
                        shape = RoundedCornerShape(12.dp),
                        modifier = Modifier
                            .fillMaxWidth()
                            .testTag("custom_strategies_empty_state")
                    ) {
                        Text(
                            text = "No strategies defined in this group yet. Create a Multiplier Interval, Time Interval, or Color Sequence strategy below.",
                            fontSize = 12.sp,
                            color = MaterialTheme.colorScheme.onSurfaceVariant,
                            modifier = Modifier.padding(12.dp)
                        )
                    }
                } else {
                    Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
                        orderedStrategies.forEachIndexed { idx, strategy ->
                            val stratOccurrences = remember(patternOccurrences, strategy.id) {
                                patternOccurrences.filter { it.strategyId == strategy.id }
                            }
                            CustomStrategyItemCard(
                                strategy = strategy,
                                occurrences = stratOccurrences,
                                canMoveUp = idx > 0,
                                canMoveDown = idx < orderedStrategies.lastIndex,
                                onMoveUp = {
                                    if (idx > 0) onReorderStrategies(strategy, orderedStrategies[idx - 1])
                                },
                                onMoveDown = {
                                    if (idx < orderedStrategies.lastIndex) onReorderStrategies(strategy, orderedStrategies[idx + 1])
                                },
                                onToggleEnabled = { onToggleCustomStrategyEnabled(strategy, it) },
                                onToggleNotifications = { onToggleCustomStrategyNotifications(strategy, it) },
                                onEdit = {
                                    editingCustomStrategy = strategy
                                    preselectedGroup = strategy.effectiveGroup()
                                    preselectedMode = strategy.strategyMode
                                    preselectedSubtype = strategy.multiplierRuleSubtype
                                    showCustomBuilder = true
                                },
                                onDelete = { onDeleteCustomStrategy(strategy.id) }
                            )
                        }
                    }
                }
            }
        }

        // Quick Strategy Templates Across All 3 Groups
        Card(
            modifier = Modifier.fillMaxWidth(),
            shape = RoundedCornerShape(16.dp),
            colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface),
            border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant)
        ) {
            Column(
                modifier = Modifier.padding(14.dp),
                verticalArrangement = Arrangement.spacedBy(10.dp)
            ) {
                Text(
                    text = "CREATE STRATEGY",
                    fontWeight = FontWeight.Bold,
                    fontSize = 12.sp,
                    color = PrimaryCobalt
                )
                FlowRow(
                    horizontalArrangement = Arrangement.spacedBy(8.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    OutlinedButton(
                        onClick = {
                            editingCustomStrategy = null
                            preselectedGroup = StrategyGroup.MULTIPLIER_INTERVAL
                            preselectedMode = StrategyMode.STRATEGY_A_CONSECUTIVE_LOWS
                            preselectedSubtype = MultiplierRuleSubtype.LOW_MULTIPLIER
                            showCustomBuilder = true
                        },
                        modifier = Modifier.testTag("new_strategy_a_btn")
                    ) {
                        Icon(Icons.Default.Add, contentDescription = null, modifier = Modifier.size(16.dp))
                        Spacer(modifier = Modifier.width(4.dp))
                        Text("A · LOW Rule", fontSize = 11.sp, fontWeight = FontWeight.Bold)
                    }
                    OutlinedButton(
                        onClick = {
                            editingCustomStrategy = null
                            preselectedGroup = StrategyGroup.MULTIPLIER_INTERVAL
                            preselectedMode = StrategyMode.STRATEGY_A_CONSECUTIVE_LOWS
                            preselectedSubtype = MultiplierRuleSubtype.ABOVE_THEN_BELOW
                            showCustomBuilder = true
                        },
                        modifier = Modifier.testTag("new_above_below_btn")
                    ) {
                        Icon(Icons.Default.Add, contentDescription = null, modifier = Modifier.size(16.dp))
                        Spacer(modifier = Modifier.width(4.dp))
                        Text("A · Above X -> Below Y", fontSize = 11.sp, fontWeight = FontWeight.Bold)
                    }
                    OutlinedButton(
                        onClick = {
                            editingCustomStrategy = null
                            preselectedGroup = StrategyGroup.TIME_INTERVAL
                            preselectedMode = StrategyMode.STRATEGY_B_BIG_INTERVALS
                            preselectedSubtype = MultiplierRuleSubtype.BIG_MULTIPLIER
                            showCustomBuilder = true
                        },
                        modifier = Modifier.testTag("new_strategy_b_btn")
                    ) {
                        Icon(Icons.Default.Add, contentDescription = null, modifier = Modifier.size(16.dp))
                        Spacer(modifier = Modifier.width(4.dp))
                        Text("B · Time Interval", fontSize = 11.sp, fontWeight = FontWeight.Bold)
                    }
                    OutlinedButton(
                        onClick = {
                            editingCustomStrategy = null
                            preselectedGroup = StrategyGroup.COLOR_SEQUENCE
                            preselectedMode = StrategyMode.STRATEGY_C_COLOR_PATTERNS
                            preselectedSubtype = MultiplierRuleSubtype.STANDARD_THRESHOLD
                            showCustomBuilder = true
                        },
                        modifier = Modifier.testTag("new_strategy_c_btn")
                    ) {
                        Icon(Icons.Default.Add, contentDescription = null, modifier = Modifier.size(16.dp))
                        Spacer(modifier = Modifier.width(4.dp))
                        Text("C · Color Sequence", fontSize = 11.sp, fontWeight = FontWeight.Bold)
                    }
                    Button(
                        onClick = {
                            editingCustomStrategy = null
                            preselectedGroup = StrategyGroup.MULTIPLIER_INTERVAL
                            preselectedMode = StrategyMode.CUSTOM_CONDITION
                            preselectedSubtype = MultiplierRuleSubtype.STANDARD_THRESHOLD
                            showCustomBuilder = true
                        },
                        modifier = Modifier.testTag("create_custom_strategy_btn")
                    ) {
                        Icon(Icons.Default.Add, contentDescription = "Custom Strategy", modifier = Modifier.size(16.dp))
                        Spacer(modifier = Modifier.width(4.dp))
                        Text("CUSTOM RULE", fontSize = 11.sp, fontWeight = FontWeight.Bold)
                    }
                }
            }
        }

        // Global Observation Classification Thresholds
        Card(
            modifier = Modifier.fillMaxWidth(),
            shape = RoundedCornerShape(16.dp),
            colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface),
            border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant)
        ) {
            Column(
                modifier = Modifier.padding(16.dp),
                verticalArrangement = Arrangement.spacedBy(10.dp)
            ) {
                Text(
                    text = "DEFAULT OBSERVATION THRESHOLDS",
                    fontWeight = FontWeight.Bold,
                    fontSize = 12.sp,
                    color = PrimaryCobalt
                )
                Text(
                    text = "Used for statistical LOW / BIG baselines across analytics views. Alerts are strictly observation-only.",
                    fontSize = 11.sp,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
                Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                    OutlinedTextField(
                        value = lowThreshText,
                        onValueChange = { lowThreshText = it },
                        label = { Text("Low Threshold (x)") },
                        singleLine = true,
                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Decimal),
                        modifier = Modifier
                            .weight(1f)
                            .testTag("strategy_low_threshold_input")
                    )
                    OutlinedTextField(
                        value = bigThreshText,
                        onValueChange = { bigThreshText = it },
                        label = { Text("Big Threshold (x)") },
                        singleLine = true,
                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Decimal),
                        modifier = Modifier
                            .weight(1f)
                            .testTag("strategy_big_threshold_input")
                    )
                }
                OutlinedTextField(
                    value = defaultConsecutiveText,
                    onValueChange = { defaultConsecutiveText = it },
                    label = { Text("Default Consecutive Low Alert Baseline") },
                    singleLine = true,
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("strategy_consecutive_baseline_input")
                )
                StrategyToggleRow(
                    label = "Live Monitoring Observation Enabled",
                    checked = monitoringEnabled,
                    testTag = "strategy_monitoring_switch"
                ) { monitoringEnabled = it }
                OutlinedTextField(
                    value = maxAttemptsText,
                    onValueChange = { maxAttemptsText = it },
                    label = { Text("Max Strategy Bet Attempts per Event") },
                    singleLine = true,
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("max_strategy_attempts_input")
                )
                StrategyToggleRow(
                    label = "Martingale Progression (follows Master Loss Progression rule)",
                    checked = martingaleEnabled,
                    testTag = "martingale_switch"
                ) { martingaleEnabled = it }

                HorizontalDivider(modifier = Modifier.padding(vertical = 4.dp))

                Text(
                    text = "MASTER LOSS PROGRESSION",
                    fontWeight = FontWeight.Bold,
                    fontSize = 12.sp,
                    color = PrimaryCobalt
                )
                Text(
                    text = "Controls post-loss stake increases across Martingale and all strategy rules. Disabled stays strictly at base stake.",
                    fontSize = 11.sp,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
                StrategyToggleRow(
                    label = "Enable Post-Loss Stake Progression",
                    checked = lossProgressionEnabled,
                    testTag = "strategy_loss_progression_switch"
                ) {
                    lossProgressionEnabled = it
                    if (it && lossProgressionMode == StakeProgressionMode.DISABLED) {
                        lossProgressionMode = StakeProgressionMode.DOUBLE_AMOUNT
                    }
                }
                FlowRow(
                    horizontalArrangement = Arrangement.spacedBy(8.dp),
                    verticalArrangement = Arrangement.spacedBy(6.dp)
                ) {
                    listOf(
                        StakeProgressionMode.DISABLED to "Disabled (Flat)",
                        StakeProgressionMode.DOUBLE_AMOUNT to "Double on Loss (2x)",
                        StakeProgressionMode.ADD_AMOUNT to "Add Amount"
                    ).forEach { (mode, title) ->
                        FilterChip(
                            selected = if (!lossProgressionEnabled) mode == StakeProgressionMode.DISABLED else lossProgressionMode == mode,
                            onClick = {
                                lossProgressionMode = mode
                                lossProgressionEnabled = mode != StakeProgressionMode.DISABLED
                            },
                            label = { Text(title, fontSize = 11.sp, fontWeight = FontWeight.Bold) },
                            modifier = Modifier.testTag("strategy_progression_mode_${mode.name}")
                        )
                    }
                }
                if (lossProgressionEnabled && lossProgressionMode == StakeProgressionMode.ADD_AMOUNT) {
                    OutlinedTextField(
                        value = lossProgressionAddAmountText,
                        onValueChange = { lossProgressionAddAmountText = it },
                        label = { Text("Add Amount on Each Loss") },
                        singleLine = true,
                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Decimal),
                        modifier = Modifier
                            .fillMaxWidth()
                            .testTag("strategy_progression_add_amount_input")
                    )
                }
            }
        }

        // Save & Close
        var isSaving by remember { mutableStateOf(false) }
        Button(
            enabled = !isSaving,
            onClick = {
                if (isSaving) return@Button
                try {
                    val parsedLow = lowThreshText.trim().toDoubleOrNull()
                    val parsedBig = bigThreshText.trim().toDoubleOrNull()
                    val parsedConsec = defaultConsecutiveText.trim().toIntOrNull()
                    val parsedMaxAttempts = maxAttemptsText.trim().toIntOrNull()?.coerceAtLeast(1) ?: 2
                    val parsedAddAmt = lossProgressionAddAmountText.trim().toDoubleOrNull() ?: config.lossProgressionAddAmount

                    if (parsedLow == null || !parsedLow.isFinite() || parsedLow < 1.01 ||
                        parsedBig == null || !parsedBig.isFinite() || parsedBig <= parsedLow ||
                        parsedConsec == null || parsedConsec < 1 ||
                        !parsedAddAmt.isFinite() || parsedAddAmt < 0.0
                    ) {
                        Toast.makeText(
                            context,
                            "Failed to save settings, please check input",
                            Toast.LENGTH_SHORT
                        ).show()
                        return@Button
                    }

                    isSaving = true
                    val effectiveMode = if (!lossProgressionEnabled) {
                        StakeProgressionMode.DISABLED
                    } else {
                        lossProgressionMode
                    }
                    val updatedConfig = config.copy(
                        lowThreshold = parsedLow,
                        bigThreshold = parsedBig,
                        strategy1LowThreshold = parsedLow,
                        strategy1RequiredLowRounds = parsedConsec,
                        strategy2HighThreshold = parsedBig,
                        monitoringEnabled = monitoringEnabled,
                        maxStrategyBetAttempts = parsedMaxAttempts,
                        isMartingaleEnabled = martingaleEnabled,
                        lossProgressionEnabled = lossProgressionEnabled && effectiveMode != StakeProgressionMode.DISABLED,
                        lossProgressionMode = effectiveMode,
                        lossProgressionAddAmount = parsedAddAmt
                    )
                    onSaveConfig(updatedConfig)
                } catch (e: Exception) {
                    isSaving = false
                    Log.e("StrategyScreen", "Save failed", e)
                    Toast.makeText(
                        context,
                        "Failed to save settings, please check input",
                        Toast.LENGTH_SHORT
                    ).show()
                }
            },
            modifier = Modifier
                .fillMaxWidth()
                .height(52.dp)
                .testTag("save_strategy_btn")
        ) {
            Icon(Icons.Default.Save, contentDescription = "Save & Close")
            Spacer(modifier = Modifier.width(8.dp))
            Text("SAVE & CLOSE", fontWeight = FontWeight.Bold)
        }
        Spacer(modifier = Modifier.height(80.dp))
    }

    if (showCustomBuilder) {
        CustomStrategyBuilderDialog(
            initial = editingCustomStrategy,
            initialGroup = editingCustomStrategy?.effectiveGroup() ?: preselectedGroup,
            initialMode = editingCustomStrategy?.strategyMode ?: preselectedMode,
            initialSubtype = editingCustomStrategy?.multiplierRuleSubtype ?: preselectedSubtype,
            existingStrategies = customStrategies,
            defaultLowThreshold = config.lowThreshold,
            defaultBigThreshold = config.bigThreshold,
            onDismiss = {
                showCustomBuilder = false
                editingCustomStrategy = null
            },
            onSave = { built ->
                val nextOrder = built.takeIf { it.id != 0L }?.sortOrder
                    ?: ((customStrategies.maxOfOrNull { it.sortOrder } ?: -1) + 1)
                onSaveCustomStrategy(built.copy(sortOrder = nextOrder))
                showCustomBuilder = false
                editingCustomStrategy = null
            }
        )
    }
    }
}

@Composable
private fun CustomStrategyItemCard(
    strategy: CustomPatternStrategy,
    occurrences: List<PatternOccurrenceRecord>,
    canMoveUp: Boolean,
    canMoveDown: Boolean,
    onMoveUp: () -> Unit,
    onMoveDown: () -> Unit,
    onToggleEnabled: (Boolean) -> Unit,
    onToggleNotifications: (Boolean) -> Unit,
    onEdit: () -> Unit,
    onDelete: () -> Unit
) {
    val statusText = when {
        !strategy.isEnabled -> "DISABLED"
        strategy.currentlyReached -> "DETECTED"
        strategy.currentlyApproaching -> "APPROACHING"
        else -> "MONITORING"
    }
    val statusColor = when {
        !strategy.isEnabled -> MaterialTheme.colorScheme.onSurfaceVariant
        strategy.currentlyReached -> ActionStartEmerald
        strategy.currentlyApproaching -> ActionCashOutAmber
        else -> PrimaryCobalt
    }

    val currentHourStart = CalendarHourUtils.hourStartFor(System.currentTimeMillis())
    val thisHourCount = occurrences.count { it.hourStartTimestamp == currentHourStart }
    val latestOcc = occurrences.maxByOrNull { it.triggerTimestampMs }

    Surface(
        shape = RoundedCornerShape(12.dp),
        color = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.35f),
        border = BorderStroke(
            width = 1.dp,
            color = if (strategy.isEnabled) statusColor.copy(alpha = 0.55f) else MaterialTheme.colorScheme.outlineVariant
        ),
        modifier = Modifier
            .fillMaxWidth()
            .testTag("custom_strategy_card_${strategy.id}")
    ) {
        Column(
            modifier = Modifier.padding(12.dp),
            verticalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Column(modifier = Modifier.weight(1f)) {
                    Text(
                        text = strategy.name,
                        fontWeight = FontWeight.Bold,
                        fontSize = 14.sp,
                        color = MaterialTheme.colorScheme.onSurface
                    )
                    Text(
                        text = "${strategy.effectiveGroup().displayTitle} · Target: ${strategy.targetGameId.uppercase(Locale.US)}",
                        fontSize = 10.sp,
                        fontWeight = FontWeight.SemiBold,
                        color = PrimaryCobalt
                    )
                    Text(
                        text = strategy.summaryDescription(),
                        fontSize = 11.sp,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }
                Switch(
                    checked = strategy.isEnabled,
                    onCheckedChange = onToggleEnabled,
                    modifier = Modifier.testTag("custom_strategy_switch_${strategy.id}")
                )
            }

            // Independent per-strategy occurrence telemetry & delivery config
            Surface(
                color = MaterialTheme.colorScheme.surface,
                shape = RoundedCornerShape(8.dp),
                modifier = Modifier.fillMaxWidth()
            ) {
                Column(modifier = Modifier.padding(8.dp), verticalArrangement = Arrangement.spacedBy(2.dp)) {
                    val repeatCapLabel = if (strategy.repeatLimit > 0) "${strategy.repeatLimit}/hr max" else "Unlimited repeats"
                    Text(
                        text = "Hour: $thisHourCount / ${strategy.requiredOccurrencesPerHour} · All-time: ${occurrences.size} · Cooldown ${strategy.repeatCooldownMinutes}m · $repeatCapLabel",
                        fontSize = 11.sp,
                        fontWeight = FontWeight.SemiBold,
                        color = MaterialTheme.colorScheme.onSurface
                    )
                    if (latestOcc != null) {
                        Text(
                            text = "Latest: ${CalendarHourUtils.formatDateTime(latestOcc.triggerTimestampMs)} (${latestOcc.gameName}) — ${latestOcc.detailsSummary}",
                            fontSize = 10.sp,
                            color = MaterialTheme.colorScheme.onSurfaceVariant
                        )
                    }
                }
            }

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Row(horizontalArrangement = Arrangement.spacedBy(6.dp), verticalAlignment = Alignment.CenterVertically) {
                    Surface(
                        color = statusColor.copy(alpha = 0.14f),
                        shape = RoundedCornerShape(6.dp)
                    ) {
                        Text(
                            text = statusText,
                            color = statusColor,
                            fontSize = 10.sp,
                            fontWeight = FontWeight.Bold,
                            modifier = Modifier.padding(horizontal = 6.dp, vertical = 2.dp)
                        )
                    }
                    Surface(
                        color = MaterialTheme.colorScheme.primaryContainer,
                        shape = RoundedCornerShape(6.dp)
                    ) {
                        Text(
                            text = if (strategy.useAlarmStyle) "ALARM" else "NOTIFY",
                            color = PrimaryCobalt,
                            fontSize = 10.sp,
                            fontWeight = FontWeight.SemiBold,
                            modifier = Modifier.padding(horizontal = 6.dp, vertical = 2.dp)
                        )
                    }
                }

                Row(verticalAlignment = Alignment.CenterVertically) {
                    if (canMoveUp) {
                        IconButton(
                            onClick = onMoveUp,
                            modifier = Modifier.testTag("strategy_move_up_${strategy.id}")
                        ) {
                            Icon(Icons.Default.ArrowUpward, contentDescription = "Move Up", tint = PrimaryCobalt, modifier = Modifier.size(18.dp))
                        }
                    }
                    if (canMoveDown) {
                        IconButton(
                            onClick = onMoveDown,
                            modifier = Modifier.testTag("strategy_move_down_${strategy.id}")
                        ) {
                            Icon(Icons.Default.ArrowDownward, contentDescription = "Move Down", tint = PrimaryCobalt, modifier = Modifier.size(18.dp))
                        }
                    }
                    IconButton(
                        onClick = { onToggleNotifications(!strategy.notificationEnabled) },
                        modifier = Modifier.testTag("custom_strategy_notif_btn_${strategy.id}")
                    ) {
                        Icon(
                            imageVector = if (strategy.notificationEnabled) Icons.Default.Notifications else Icons.Default.NotificationsOff,
                            contentDescription = "Toggle Strategy Notifications",
                            tint = if (strategy.notificationEnabled) ActionStartEmerald else MaterialTheme.colorScheme.onSurfaceVariant,
                            modifier = Modifier.size(18.dp)
                        )
                    }
                    IconButton(
                        onClick = onEdit,
                        modifier = Modifier.testTag("strategy_edit_btn_${strategy.id}")
                    ) {
                        Icon(
                            imageVector = Icons.Default.Edit,
                            contentDescription = "Edit Custom Strategy",
                            tint = PrimaryCobalt,
                            modifier = Modifier.size(18.dp)
                        )
                    }
                    IconButton(
                        onClick = onDelete,
                        modifier = Modifier.testTag("strategy_delete_btn_${strategy.id}")
                    ) {
                        Icon(
                            imageVector = Icons.Default.Delete,
                            contentDescription = "Delete Custom Strategy",
                            tint = ActionStopRed,
                            modifier = Modifier.size(18.dp)
                        )
                    }
                }
            }
        }
    }
}

@OptIn(ExperimentalLayoutApi::class)
@Composable
private fun CustomStrategyBuilderDialog(
    initial: CustomPatternStrategy?,
    initialGroup: StrategyGroup,
    initialMode: StrategyMode,
    initialSubtype: MultiplierRuleSubtype,
    existingStrategies: List<CustomPatternStrategy>,
    defaultLowThreshold: Double,
    defaultBigThreshold: Double,
    onDismiss: () -> Unit,
    onSave: (CustomPatternStrategy) -> Unit
) {
    val context = LocalContext.current

    var strategyGroup by remember { mutableStateOf(initial?.effectiveGroup() ?: initialGroup) }
    var strategyMode by remember { mutableStateOf(initial?.strategyMode ?: initialMode) }
    var multiplierSubtype by remember { mutableStateOf(initial?.multiplierRuleSubtype ?: initialSubtype) }

    var name by remember {
        mutableStateOf(
            initial?.name ?: when {
                initialSubtype == MultiplierRuleSubtype.ABOVE_THEN_BELOW -> "Above X (3x) -> Below X (2x) Ring"
                initialGroup == StrategyGroup.TIME_INTERVAL -> "Time Interval · Wait Before Bet"
                initialGroup == StrategyGroup.COLOR_SEQUENCE -> "Color Sequence Rule"
                initialSubtype == MultiplierRuleSubtype.LOW_MULTIPLIER -> "LOW Multiplier Consecutive"
                else -> "Custom Multiplier Strategy"
            }
        )
    }
    var targetGameId by remember { mutableStateOf(initial?.targetGameId ?: "ALL") }
    var analysisWindowMinutes by remember { mutableStateOf(initial?.analysisWindowMinutes ?: 60) }
    var customWindowMinutesText by remember { mutableStateOf((initial?.analysisWindowMinutes ?: 60).toString()) }
    var requiredHourlyOccText by remember { mutableStateOf((initial?.requiredOccurrencesPerHour ?: 2).toString()) }
    var inclusiveBounds by remember { mutableStateOf(initial?.inclusiveBounds ?: false) }

    // Multiplier Interval fields
    val initCond = initial?.getConditions()?.firstOrNull()
    var aboveThresholdXText by remember { mutableStateOf(String.format(Locale.US, "%.2f", initial?.aboveThresholdX ?: defaultLowThreshold)) }
    var aboveCountText by remember { mutableStateOf((initial?.aboveConsecutiveCount ?: 3).toString()) }
    var belowThresholdYText by remember { mutableStateOf(String.format(Locale.US, "%.2f", initial?.belowThresholdY ?: defaultLowThreshold)) }
    var belowCountText by remember { mutableStateOf((initial?.belowConsecutiveCount ?: 2).toString()) }
    var logicOperator by remember { mutableStateOf(initial?.logicOperator ?: PatternLogicOperator.AND) }
    var conditions by remember {
        mutableStateOf(
            initial?.getConditions()?.takeIf { it.isNotEmpty() }
                ?: listOf(
                    PatternCondition(
                        comparator = if (initialSubtype == MultiplierRuleSubtype.BIG_MULTIPLIER) PatternComparator.GREATER_OR_EQUAL else PatternComparator.LESS_THAN,
                        threshold = if (initialSubtype == MultiplierRuleSubtype.BIG_MULTIPLIER) defaultBigThreshold else defaultLowThreshold,
                        consecutiveRounds = if (initialSubtype == MultiplierRuleSubtype.BIG_MULTIPLIER) 1 else 6
                    )
                )
        )
    }

    // Time Interval fields
    var bigMultThreshText by remember {
        mutableStateOf(
            String.format(
                Locale.US,
                "%.2f",
                initial?.bigMultiplierThreshold ?: initCond?.threshold ?: defaultBigThreshold
            )
        )
    }
    var postBigWaitText by remember { mutableStateOf((initial?.postBigWaitMinutes ?: 5).toString()) }
    var specificAlertTimesCsv by remember { mutableStateOf(initial?.specificAlertTimesCsv ?: "") }
    var minGapText by remember { mutableStateOf((initial?.minIntervalMinutes ?: 5).toString()) }
    var maxGapText by remember { mutableStateOf((initial?.maxIntervalMinutes ?: 15).toString()) }

    // Color Sequence fields
    var colorDirection by remember { mutableStateOf(initial?.colorMatchDirection ?: ColorMatchDirection.HORIZONTAL) }
    var gridColsVertical by remember { mutableStateOf(initial?.gridColumnsForVertical ?: 8) }
    var colorSequence by remember {
        mutableStateOf(
            initial?.getColorSequence()?.takeIf { it.isNotEmpty() }
                ?: listOf(CrashColorCategory.Purple, CrashColorCategory.Purple, CrashColorCategory.Blue)
        )
    }
    var customHexCsv by remember { mutableStateOf(initial?.customColorHexFilterCsv ?: "") }

    // Notifications, Cooldown & Repeats
    var waitingPeriodText by remember { mutableStateOf((initial?.waitingPeriodSeconds ?: 0).toString()) }
    var priority by remember { mutableStateOf(initial?.notificationLevel ?: NotificationPriorityLevel.HIGH) }
    var approachingEnabled by remember { mutableStateOf(initial?.approachingEnabled ?: true) }
    var notificationEnabled by remember { mutableStateOf(initial?.notificationEnabled ?: true) }
    var useAlarmStyle by remember { mutableStateOf(initial?.useAlarmStyle ?: true) }
    var vibrationEnabled by remember { mutableStateOf(initial?.vibrationEnabled ?: true) }
    var customMessage by remember { mutableStateOf(initial?.customMessage ?: "") }
    var repeatPolicy by remember { mutableStateOf(initial?.repeatPolicy ?: RepeatNotificationPolicy.ONCE_PER_OCCURRENCE) }
    var cooldownMinutesText by remember { mutableStateOf((initial?.repeatCooldownMinutes ?: 0).toString()) }
    var repeatLimitText by remember { mutableStateOf((initial?.repeatLimit ?: 0).toString()) }

    // Review state
    var reviewCandidate by remember { mutableStateOf<CustomPatternStrategy?>(null) }

    if (LocalOnBackPressedDispatcherOwner.current != null) {
        BackHandler {
            if (reviewCandidate != null) reviewCandidate = null else onDismiss()
        }
    }

    val selectableColors = remember {
        listOf(
            CrashColorCategory.Blue,
            CrashColorCategory.Purple,
            CrashColorCategory.Pink,
            CrashColorCategory.Red,
            CrashColorCategory.Green,
            CrashColorCategory.Orange
        )
    }

    fun tryBuildCandidate(): Pair<CustomPatternStrategy?, String?> {
        val trimmedName = name.trim()
        val waitSec = waitingPeriodText.trim().toIntOrNull()?.coerceAtLeast(0)
        val reqOcc = requiredHourlyOccText.trim().toIntOrNull()?.coerceAtLeast(1)
        val cooldownMin = cooldownMinutesText.trim().toIntOrNull()?.coerceAtLeast(0) ?: 0
        val repLimit = repeatLimitText.trim().toIntOrNull()?.coerceAtLeast(0) ?: 0
        val winMin = (if (analysisWindowMinutes <= 0) customWindowMinutesText.trim().toIntOrNull() else analysisWindowMinutes)
            ?.coerceAtLeast(5)

        if (trimmedName.isEmpty() || waitSec == null || reqOcc == null || winMin == null) {
            return null to "Failed to save settings, please check input"
        }

        val finalConditions: List<PatternCondition>
        var resolvedMode = strategyMode
        var aboveX = aboveThresholdXText.trim().toDoubleOrNull() ?: defaultLowThreshold
        var aboveN = aboveCountText.trim().toIntOrNull() ?: 3
        var belowY = belowThresholdYText.trim().toDoubleOrNull() ?: defaultLowThreshold
        var belowM = belowCountText.trim().toIntOrNull() ?: 2
        var bigTh = bigMultThreshText.trim().toDoubleOrNull() ?: defaultBigThreshold
        var postWait = postBigWaitText.trim().toIntOrNull() ?: 0
        var minGap = minGapText.trim().toIntOrNull() ?: 5
        var maxGap = maxGapText.trim().toIntOrNull() ?: 15
        var colorCsv = ""

        when (strategyGroup) {
            StrategyGroup.MULTIPLIER_INTERVAL -> {
                if (multiplierSubtype == MultiplierRuleSubtype.ABOVE_THEN_BELOW) {
                    if (aboveX < 1.0 || belowY < 1.0 || aboveN < 1 || belowM < 1) {
                        return null to "Enter valid thresholds (>= 1.00) and consecutive counts (>= 1)."
                    }
                    resolvedMode = StrategyMode.STRATEGY_A_CONSECUTIVE_LOWS
                    finalConditions = listOf(
                        PatternCondition(
                            comparator = if (inclusiveBounds) PatternComparator.GREATER_OR_EQUAL else PatternComparator.GREATER_THAN,
                            threshold = aboveX,
                            consecutiveRounds = aboveN,
                            timeWindowMinutes = winMin
                        ),
                        PatternCondition(
                            comparator = if (inclusiveBounds) PatternComparator.LESS_OR_EQUAL else PatternComparator.LESS_THAN,
                            threshold = belowY,
                            consecutiveRounds = belowM,
                            timeWindowMinutes = winMin
                        )
                    )
                } else {
                    if (conditions.isEmpty()) {
                        return null to "Add at least one multiplier condition."
                    }
                    resolvedMode = if (strategyMode == StrategyMode.CUSTOM_CONDITION) StrategyMode.CUSTOM_CONDITION else StrategyMode.STRATEGY_A_CONSECUTIVE_LOWS
                    finalConditions = conditions
                }
            }

            StrategyGroup.TIME_INTERVAL -> {
                if (bigTh < 1.01 || postWait < 0 || minGap < 0 || maxGap < minGap) {
                    return null to "Check BIG threshold (>= 1.01), wait minutes (>= 0), and gap range."
                }
                resolvedMode = StrategyMode.STRATEGY_B_BIG_INTERVALS
                finalConditions = listOf(
                    PatternCondition(
                        comparator = if (inclusiveBounds) PatternComparator.GREATER_OR_EQUAL else PatternComparator.GREATER_THAN,
                        threshold = bigTh,
                        consecutiveRounds = 1,
                        timeWindowMinutes = winMin
                    )
                )
            }

            StrategyGroup.COLOR_SEQUENCE -> {
                if (colorSequence.isEmpty() && customHexCsv.isBlank()) {
                    return null to "Tap at least one color or specify custom HEX colors."
                }
                resolvedMode = StrategyMode.STRATEGY_C_COLOR_PATTERNS
                colorCsv = colorSequence.joinToString(",") { it.name }
                finalConditions = listOf(
                    PatternCondition(
                        comparator = PatternComparator.LESS_THAN,
                        threshold = 2.00,
                        consecutiveRounds = maxOf(colorSequence.size, 1),
                        timeWindowMinutes = winMin
                    )
                )
            }
        }

        val built = (initial ?: CustomPatternStrategy(name = trimmedName))
            .copy(
                name = trimmedName,
                isEnabled = initial?.isEnabled ?: true,
                strategyGroup = strategyGroup,
                strategyMode = resolvedMode,
                multiplierRuleSubtype = multiplierSubtype,
                logicOperator = logicOperator,
                aboveThresholdX = aboveX,
                aboveConsecutiveCount = aboveN,
                belowThresholdY = belowY,
                belowConsecutiveCount = belowM,
                inclusiveBounds = inclusiveBounds,
                postBigWaitMinutes = postWait,
                bigMultiplierThreshold = bigTh,
                specificAlertTimesCsv = specificAlertTimesCsv.trim(),
                colorMatchDirection = colorDirection,
                gridColumnsForVertical = gridColsVertical,
                customColorHexFilterCsv = customHexCsv.trim(),
                waitingPeriodSeconds = waitSec,
                notificationLevel = priority,
                approachingEnabled = approachingEnabled,
                customMessage = customMessage.trim(),
                repeatPolicy = repeatPolicy,
                repeatCooldownMinutes = cooldownMin,
                repeatLimit = repLimit,
                notificationEnabled = notificationEnabled,
                useAlarmStyle = useAlarmStyle,
                vibrationEnabled = vibrationEnabled,
                requiredOccurrencesPerHour = reqOcc,
                analysisWindowMinutes = winMin,
                minIntervalMinutes = minGap,
                maxIntervalMinutes = maxGap,
                colorSequenceCsv = colorCsv,
                targetGameId = targetGameId
            )
            .withConditions(finalConditions)
        return built to null
    }

    val livePreviewValidation = remember(
        strategyGroup, multiplierSubtype, name, targetGameId, analysisWindowMinutes,
        aboveThresholdXText, aboveCountText, belowThresholdYText, belowCountText,
        inclusiveBounds, bigMultThreshText, postBigWaitText, minGapText, maxGapText,
        colorSequence, customHexCsv, conditions, customWindowMinutesText
    ) {
        tryBuildCandidate().first?.let { candidate ->
            StrategyRangeValidator.validate(candidate, existingStrategies, defaultLowThreshold, defaultBigThreshold)
        }
    }

    val hasActivityRegistry = LocalActivityResultRegistryOwner.current != null
    val builderCardContent: @Composable () -> Unit = {
        Surface(
            shape = RoundedCornerShape(18.dp),
            color = MaterialTheme.colorScheme.surface,
            tonalElevation = 6.dp,
            modifier = Modifier
                .fillMaxWidth()
                .heightIn(max = 680.dp)
        ) {
            val activeReview = reviewCandidate
            if (activeReview != null) {
                // Step 2: Explicit Pre-Save Review Screen
                val validation = StrategyRangeValidator.validate(
                    activeReview,
                    existingStrategies,
                    defaultLowThreshold,
                    defaultBigThreshold
                )
                Column(
                    modifier = Modifier
                        .verticalScroll(rememberScrollState())
                        .padding(16.dp)
                        .testTag("strategy_review_panel"),
                    verticalArrangement = Arrangement.spacedBy(10.dp)
                ) {
                    Text(
                        text = "REVIEW STRATEGY BEFORE SAVING",
                        fontWeight = FontWeight.ExtraBold,
                        fontSize = 15.sp,
                        color = PrimaryCobalt
                    )
                    Surface(
                        color = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.45f),
                        shape = RoundedCornerShape(12.dp),
                        modifier = Modifier.fillMaxWidth()
                    ) {
                        Column(modifier = Modifier.padding(12.dp), verticalArrangement = Arrangement.spacedBy(5.dp)) {
                            Text("Name: ${activeReview.name}", fontWeight = FontWeight.Bold, fontSize = 13.sp)
                            Text("Group: ${activeReview.effectiveGroup().displayTitle}", fontSize = 12.sp)
                            Text("Target Game: ${activeReview.targetGameId}", fontSize = 12.sp)
                            Text("Condition Summary: ${activeReview.summaryDescription()}", fontSize = 12.sp, fontWeight = FontWeight.SemiBold, color = PrimaryCobalt)
                            Text("Boundary Mode: ${if (activeReview.inclusiveBounds) "Inclusive (>= / <=)" else "Exclusive (> / <)"}", fontSize = 11.sp)
                            Text("Analysis Window: ${activeReview.analysisWindowMinutes}m · Min Occurrences: ${activeReview.requiredOccurrencesPerHour}", fontSize = 11.sp)
                            Text("Alert Style: ${if (activeReview.useAlarmStyle) "Alarm" else "Normal Notification"} · Vibration: ${if (activeReview.vibrationEnabled) "ON" else "OFF"}", fontSize = 11.sp)
                            Text("Cooldown: ${activeReview.repeatCooldownMinutes}m · Repeat Limit: ${if (activeReview.repeatLimit == 0) "Unlimited" else "${activeReview.repeatLimit}/hr"}", fontSize = 11.sp)
                            if (activeReview.specificAlertTimesCsv.isNotBlank()) {
                                Text("Specific Alert Times: ${activeReview.specificAlertTimesCsv}", fontSize = 11.sp)
                            }
                            if (activeReview.customColorHexFilterCsv.isNotBlank()) {
                                Text("Custom HEX Colors: ${activeReview.customColorHexFilterCsv}", fontSize = 11.sp)
                            }
                        }
                    }

                    if (validation.blockingErrors.isNotEmpty() || validation.warnings.isNotEmpty()) {
                        Surface(
                            color = if (validation.blockingErrors.isNotEmpty()) Color(0xFFFEE2E2) else Color(0xFFFEF3C7),
                            shape = RoundedCornerShape(10.dp),
                            modifier = Modifier.fillMaxWidth()
                        ) {
                            Column(modifier = Modifier.padding(10.dp), verticalArrangement = Arrangement.spacedBy(4.dp)) {
                                validation.blockingErrors.forEach { err ->
                                    Text("Error: $err", color = ActionStopRed, fontSize = 11.sp, fontWeight = FontWeight.Bold)
                                }
                                validation.warnings.forEach { warn ->
                                    Text("Warning: $warn", color = Color(0xFF92400E), fontSize = 11.sp, fontWeight = FontWeight.SemiBold)
                                }
                            }
                        }
                    }

                    Row(horizontalArrangement = Arrangement.spacedBy(8.dp), modifier = Modifier.fillMaxWidth()) {
                        OutlinedButton(
                            onClick = { reviewCandidate = null },
                            modifier = Modifier.weight(1f)
                        ) {
                            Text("EDIT SETTINGS")
                        }
                        Button(
                            enabled = validation.isValid,
                            onClick = { onSave(activeReview) },
                            modifier = Modifier
                                .weight(1f)
                                .testTag("builder_save_strategy_btn")
                        ) {
                            Text("CONFIRM SAVE", fontWeight = FontWeight.Bold)
                        }
                    }
                }
            } else {
                // Step 1: Configuration Form
                Column(
                    modifier = Modifier
                        .verticalScroll(rememberScrollState())
                        .padding(16.dp),
                    verticalArrangement = Arrangement.spacedBy(10.dp)
                ) {
                    Text(
                        text = if (initial == null) "CREATE STRATEGY" else "EDIT STRATEGY",
                        fontWeight = FontWeight.Bold,
                        fontSize = 15.sp,
                        color = PrimaryCobalt
                    )

                    Text("Strategy Group", fontSize = 12.sp, fontWeight = FontWeight.SemiBold)
                    FlowRow(horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                        StrategyGroup.entries.forEach { grp ->
                            FilterChip(
                                selected = strategyGroup == grp,
                                onClick = {
                                    strategyGroup = grp
                                    strategyMode = when (grp) {
                                        StrategyGroup.MULTIPLIER_INTERVAL -> StrategyMode.STRATEGY_A_CONSECUTIVE_LOWS
                                        StrategyGroup.TIME_INTERVAL -> StrategyMode.STRATEGY_B_BIG_INTERVALS
                                        StrategyGroup.COLOR_SEQUENCE -> StrategyMode.STRATEGY_C_COLOR_PATTERNS
                                    }
                                },
                                label = { Text(grp.displayTitle, fontSize = 10.sp) },
                                modifier = Modifier.testTag("builder_group_${grp.name}")
                            )
                        }
                    }

                    OutlinedTextField(
                        value = name,
                        onValueChange = { name = it },
                        label = { Text("Strategy Name") },
                        singleLine = true,
                        modifier = Modifier
                            .fillMaxWidth()
                            .testTag("builder_strategy_name_input")
                    )

                    Text("Target Game", fontSize = 12.sp, fontWeight = FontWeight.SemiBold)
                    FlowRow(horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                        FilterChip(
                            selected = targetGameId == "ALL",
                            onClick = { targetGameId = "ALL" },
                            label = { Text("All Games", fontSize = 11.sp) }
                        )
                        FilterChip(
                            selected = targetGameId == "unknown",
                            onClick = { targetGameId = "unknown" },
                            label = { Text("Unknown", fontSize = 11.sp) }
                        )
                        GameAdapterRegistry.allAdapters().forEach { adapter ->
                            FilterChip(
                                selected = targetGameId == adapter.gameId,
                                onClick = { targetGameId = adapter.gameId },
                                label = { Text(adapter.displayedGameName, fontSize = 11.sp) }
                            )
                        }
                    }

                    when (strategyGroup) {
                        StrategyGroup.MULTIPLIER_INTERVAL -> {
                            Text("Condition Type", fontSize = 12.sp, fontWeight = FontWeight.SemiBold)
                            FlowRow(horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                                MultiplierRuleSubtype.entries.forEach { sub ->
                                    FilterChip(
                                        selected = multiplierSubtype == sub,
                                        onClick = { multiplierSubtype = sub },
                                        label = { Text(sub.label, fontSize = 10.sp) },
                                        modifier = Modifier.testTag("builder_subtype_${sub.name}")
                                    )
                                }
                            }

                            StrategyToggleRow(
                                label = "Inclusive Boundaries (>= / <= instead of > / <)",
                                checked = inclusiveBounds,
                                testTag = "builder_inclusive_switch"
                            ) { inclusiveBounds = it }

                            if (multiplierSubtype == MultiplierRuleSubtype.ABOVE_THEN_BELOW) {
                                Text(
                                    text = "Trigger when ABOVE X occurs N times in a row, immediately followed by BELOW Y occurring M times in a row:",
                                    fontSize = 11.sp,
                                    color = MaterialTheme.colorScheme.onSurfaceVariant
                                )
                                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                                    OutlinedTextField(
                                        value = aboveThresholdXText,
                                        onValueChange = { aboveThresholdXText = it },
                                        label = { Text("Above Threshold X") },
                                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Decimal),
                                        singleLine = true,
                                        modifier = Modifier.weight(1f).testTag("builder_above_x_input")
                                    )
                                    OutlinedTextField(
                                        value = aboveCountText,
                                        onValueChange = { aboveCountText = it },
                                        label = { Text("Above Count (e.g. 3)") },
                                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                                        singleLine = true,
                                        modifier = Modifier.weight(1f).testTag("builder_above_count_input")
                                    )
                                }
                                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                                    OutlinedTextField(
                                        value = belowThresholdYText,
                                        onValueChange = { belowThresholdYText = it },
                                        label = { Text("Below Threshold Y") },
                                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Decimal),
                                        singleLine = true,
                                        modifier = Modifier.weight(1f).testTag("builder_below_y_input")
                                    )
                                    OutlinedTextField(
                                        value = belowCountText,
                                        onValueChange = { belowCountText = it },
                                        label = { Text("Below Count (e.g. 2)") },
                                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                                        singleLine = true,
                                        modifier = Modifier.weight(1f).testTag("builder_below_count_input")
                                    )
                                }
                            } else {
                                Text("Condition Logic", fontSize = 12.sp, fontWeight = FontWeight.SemiBold)
                                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                                    PatternLogicOperator.entries.forEach { op ->
                                        FilterChip(
                                            selected = logicOperator == op,
                                            onClick = { logicOperator = op },
                                            label = { Text(op.name, fontSize = 11.sp, fontWeight = FontWeight.Bold) }
                                        )
                                    }
                                }
                                conditions.forEachIndexed { idx, cond ->
                                    ConditionEditorBlock(
                                        index = idx,
                                        condition = cond,

/**

data class StakeInstruction(
    val stakeAmount: Double = 10.0,
    val formattedStake: String = "10",
    val progressionMode: StakeProgressionMode = StakeProgressionMode.DISABLED,
    val incrementAmount: Double = 10.0,

    val stakeLevel: Int = 1,

class StrategyEngine {

    companion object {
        private const val TAG = "StrategyEngine"
        const val DEFAULT_SESSION_TIMEOUT_MS = 180_000L // 3-minute guard

        private val _actionEvent = MutableStateFlow<AutomationActionEvent?>(null)
        val actionEvent: StateFlow<AutomationActionEvent?> = _actionEvent.asStateFlow()

        private val _latestStakeInstruction = MutableStateFlow(StakeInstruction())
        val latestStakeInstruction: StateFlow<StakeInstruction> = _latestStakeInstruction.asStateFlow()

        fun triggerAction(event: AutomationActionEvent) {
            _actionEvent.value = event
            safeLogI("Automation Action Triggered: ${event.actionType} (${event.amountOrMultiplier}) - ${event.reason}")
        }

        fun roundStake(value: Double): Double {
            return kotlin.math.round(value.coerceAtLeast(0.0) * 100.0) / 100.0
        }

        fun formatStakeAmount(stake: Double): String {
            val rounded = roundStake(stake)
            return if (rounded % 1.0 == 0.0) {
                rounded.toLong().toString()
            } else {
                String.format(Locale.US, "%.2f", rounded)
            }
        }

        private fun safeLogI(message: String) {
            try {
                Log.i(TAG, message)
            } catch (_: RuntimeException) {
                // Unit test JVM without Android Log mock
            }
        }

        private fun safeLogW(message: String) {
            try {
                Log.w(TAG, message)
            } catch (_: RuntimeException) {
                // Unit test JVM without Android Log mock
            }
        }
    }

    private var state = StrategyState()
    private var currentStakeInstruction = StakeInstruction()
    private val _boundConfig = MutableStateFlow(StrategyConfig())
    val boundConfig: StateFlow<StrategyConfig> = _boundConfig.asStateFlow()

    private val highMultiplierLossTimestamps = mutableListOf<Long>()
    private var totalRounds: Int = 0
    private var totalWins: Int = 0
    private var totalLosses: Int = 0

    fun getState(): StrategyState = state

    fun getPendingStakeInstruction(): StakeInstruction = currentStakeInstruction

    /**
     * Binds the active StrategyConfig selections (from Settings or Strategy tab) directly to StrategyEngine
     * so single, multi, or all-strategy selections take effect immediately.
     */
    private fun resolveEffectiveProgression(config: StrategyConfig): Pair<StakeProgressionMode, Double> {
        val masterMode = config.resolvedMasterProgressionMode()
        return if (masterMode == StakeProgressionMode.DISABLED) {
            StakeProgressionMode.DISABLED to 0.0
        } else {
            masterMode to roundStake(config.lossProgressionAddAmount.coerceAtLeast(0.0))
        }
    }

    fun bindStrategyConfig(config: StrategyConfig) {
        _boundConfig.value = config
        val enabledIds = config.getRuleConfigs()
            .filter { it.isEnabled }
            .map { it.ruleId }
            .toSet()

        val shouldClearPendingWait = state.pendingWaitRuleId != null && state.pendingWaitRuleId !in enabledIds
        val updatedStake = if (!state.isBettingSessionActive) roundStake(config.baseStake) else state.currentCalculatedStake
        val (masterMode, masterAddAmount) = resolveEffectiveProgression(config)

        state = state.copy(
            enabledRuleIds = enabledIds,
            pendingWaitRuleId = if (shouldClearPendingWait) null else state.pendingWaitRuleId,
            pendingWaitRuleLabel = if (shouldClearPendingWait) null else state.pendingWaitRuleLabel,
            waitUntilTimestamp = if (shouldClearPendingWait) 0L else state.waitUntilTimestamp,
            activeWaitRemainingMs = if (shouldClearPendingWait) 0L else state.activeWaitRemainingMs,
            activeProgressionMode = masterMode,
            activeCustomAddAmount = masterAddAmount,
            currentCalculatedStake = updatedStake
        )

        if (!state.isBettingSessionActive) {
            publishStakeInstruction(
                stake = updatedStake,
                mode = masterMode,
                addAmount = masterAddAmount,
                targetMultiplier = config.targetMultiplier,
                stakeLevel = 1,
                isSessionActive = false,
                reason = if (enabledIds.isEmpty()) "No Recovery Rule Selected" else "${enabledIds.size} Rules Bound"
            )
        }
        safeLogI("Bound ${enabledIds.size} strategy rules to StrategyEngine: $enabledIds (progression=$masterMode)")
    }

    fun updateStrategySelection(
        selectedRuleIds: Set<StrategyRuleId>,
        isStrategy1Enabled: Boolean? = null,
        isStrategy2Enabled: Boolean? = null,
        isHighMultiplierLossRuleEnabled: Boolean? = null
    ): StrategyConfig {
        val current = _boundConfig.value
        val updatedRules = current.getRuleConfigs().map { rule ->
            rule.copy(isEnabled = rule.ruleId in selectedRuleIds)
        }
        val updatedConfig = current.copy(
            isStrategy1Enabled = isStrategy1Enabled ?: current.isStrategy1Enabled,
            isStrategy2Enabled = isStrategy2Enabled ?: current.isStrategy2Enabled,
            isHighMultiplierLossRuleEnabled = isHighMultiplierLossRuleEnabled ?: current.isHighMultiplierLossRuleEnabled
        ).withUpdatedRuleConfigs(updatedRules)
        bindStrategyConfig(updatedConfig)
        return updatedConfig
    }

    /**
     * Core stake calculation after a loss event strictly respecting master StakeProgressionMode
     * (DISABLED stays flat, ADD_AMOUNT adds configured amount, DOUBLE_AMOUNT doubles 2x).
     * Never silently doubles or overrides user settings.
     */
    fun calculateNextStakeAfterLoss(
        currentStake: Double,
        progressionMode: StakeProgressionMode,
        customAddAmount: Double,
        lossMultiplier: Double = 0.0,
        isHighMultiplierLossRuleEnabled: Boolean = false
    ): Double {
        val safeCurrent = currentStake.coerceAtLeast(0.0)
        val rawNext = when (progressionMode) {
            StakeProgressionMode.DISABLED -> safeCurrent
            StakeProgressionMode.ADD_AMOUNT -> safeCurrent + customAddAmount.coerceAtLeast(0.0)
            StakeProgressionMode.DOUBLE_AMOUNT -> safeCurrent * 2.0
        }
        return roundStake(rawNext)
    }

    private fun publishStakeInstruction(
        stake: Double,
        mode: StakeProgressionMode,
        addAmount: Double,
        targetMultiplier: Double,
        stakeLevel: Int,
        isSessionActive: Boolean,
        reason: String
    ) {
        val rounded = roundStake(stake)
        val instruction = StakeInstruction(
            stakeAmount = rounded,
            formattedStake = formatStakeAmount(rounded),
            progressionMode = mode,
            incrementAmount = roundStake(addAmount),
            targetMultiplier = targetMultiplier,
            stakeLevel = stakeLevel,
            isSessionActive = isSessionActive,
            reason = reason
        )
        currentStakeInstruction = instruction
        _latestStakeInstruction.value = instruction
    }

    /**
     * Checks:
     * 1. 3-minute active session timeout guard: if an active betting session exceeds
     *    config.sessionTimeoutMinutes (default 3m), gracefully terminates the session and resets counters.
     * 2. Pending multiplier-range wait timers (e.g., 5m or 10m) — starts recovery ONLY when unlocked.
     */
    fun checkPendingTimers(
        config: StrategyConfig = _boundConfig.value,
        currentTimeMs: Long = System.currentTimeMillis()
    ): StrategyTrigger {
        val timeoutLimitMs = config.sessionTimeoutMinutes.coerceAtLeast(1) * 60_000L

        // 1. Enforce 3-minute grace period & auto-cancellation on active session
        if (state.isBettingSessionActive && state.sessionStartedTimestamp > 0L) {
            val elapsed = (currentTimeMs - state.sessionStartedTimestamp).coerceAtLeast(0L)
            if (elapsed >= timeoutLimitMs) {
                val timedOutLabel = state.activeRuleLabel ?: state.activeRuleId?.displayName ?: "Active Session"
                safeLogW("3-minute timeout guard triggered for '$timedOutLabel' after ${elapsed}ms. Terminating cycle.")
                val timeoutTrigger = StrategyTrigger.SessionTimedOut(timedOutLabel, elapsed)
                state = state.copy(
                    isBettingSessionActive = false,
                    sessionStartedTimestamp = 0L,
                    sessionTimeoutRemainingMs = 0L,
                    activeRuleId = null,
                    activeRuleLabel = null,
                    activeTrigger = timeoutTrigger,
                    currentStakeLevel = 1,
                    consecutiveWinsInSession = 0,
                    consecutiveLossesInSession = 0,
                    consecutiveObservedLosses = 0,
                    currentCalculatedStake = config.baseStake,
                    consecutiveLowCount = 0,
                    postHighWaitCount = 0
                )
                publishStakeInstruction(
                    stake = config.baseStake,
                    mode = state.activeProgressionMode,
                    addAmount = state.activeCustomAddAmount,
                    targetMultiplier = config.targetMultiplier,
                    stakeLevel = 1,
                    isSessionActive = false,
                    reason = "3m Guard Timed Out -> Reset to Base Stake"
                )
                return timeoutTrigger
            } else {
                state = state.copy(sessionTimeoutRemainingMs = timeoutLimitMs - elapsed)
            }
        }

        // 2. Exclusive lock: if a strategy session is already running, wait before firing another
        if (state.isBettingSessionActive) {
            return StrategyTrigger.None
        }

        // 3. Check multiplier-range wait timer
        val pendingRuleId = state.pendingWaitRuleId ?: return StrategyTrigger.None
        val waitUntil = state.waitUntilTimestamp
        if (waitUntil <= 0L) return StrategyTrigger.None

        val remaining = (waitUntil - currentTimeMs).coerceAtLeast(0L)
        if (remaining > 0L) {
            state = state.copy(activeWaitRemainingMs = remaining)
            return StrategyTrigger.None
        }

        val ruleConfig = config.getRuleConfig(pendingRuleId)
        val trigger = StrategyTrigger.WaitTimerCompleted(ruleConfig)
        val initialStake = roundStake(config.baseStake)
        val (effectiveMode, effectiveAddAmount) = resolveEffectiveProgression(config)

        state = state.copy(
            pendingWaitRuleId = null,
            pendingWaitRuleLabel = null,
            waitUntilTimestamp = 0L,
            activeWaitRemainingMs = 0L,
            activeRuleId = pendingRuleId,
            activeRuleLabel = ruleConfig.dynamicTitle(),
            activeProgressionMode = effectiveMode,
            activeCustomAddAmount = effectiveAddAmount,
            targetWinsToFetch = ruleConfig.targetWinsToFetch,
            activeTrigger = trigger,
            isBettingSessionActive = true,
            sessionStartedTimestamp = currentTimeMs,
            sessionTimeoutRemainingMs = timeoutLimitMs,
            currentCalculatedStake = initialStake,
            consecutiveWinsInSession = 0,
            consecutiveLossesInSession = 0
        )

        publishStakeInstruction(
            stake = initialStake,
            mode = effectiveMode,
            addAmount = effectiveAddAmount,
            targetMultiplier = config.targetMultiplier,
            stakeLevel = 1,
            isSessionActive = true,
            reason = "${ruleConfig.dynamicTitle()} Ready"
        )

        triggerAction(
            AutomationActionEvent(
                actionType = ActionType.BET,
                amountOrMultiplier = initialStake,
                reason = "${ruleConfig.dynamicTitle()} wait elapsed -> Fetch ${ruleConfig.targetWinsToFetch} wins",
                timestamp = currentTimeMs
            )
        )
        return trigger
    }

    /**
     * Activates a betting recovery session when an external CustomPatternStrategy rule fires,
     * respecting the exclusive lock (no-op if a session is already active).
     */
    fun activateExternalPatternTrigger(
        strategyName: String,
        config: StrategyConfig = _boundConfig.value,
        currentTimeMs: Long = System.currentTimeMillis(),
        progressionModeOverride: StakeProgressionMode? = null,
        customAddAmountOverride: Double? = null,
        targetWinsOverride: Int? = null,
        targetMultiplierOverride: Double? = null
    ): Boolean {
        if (state.isBettingSessionActive) return false
        val firstEnabledRule = config.getRuleConfigs().firstOrNull { it.isEnabled }
        val (mode, addAmt) = resolveEffectiveProgression(config)
        val targetWins = (targetWinsOverride ?: firstEnabledRule?.targetWinsToFetch ?: config.stopAfterConsecutiveWins).coerceAtLeast(1)
        val effectiveTargetMult = (targetMultiplierOverride ?: config.targetMultiplier).coerceAtLeast(1.01)
        val initialStake = roundStake(config.baseStake)
        val timeoutLimitMs = config.sessionTimeoutMinutes.coerceAtLeast(1) * 60_000L

        state = state.copy(
            isBettingSessionActive = true,
            activeRuleId = null,
            activeRuleLabel = strategyName,
            activeProgressionMode = mode,
            activeCustomAddAmount = addAmt,
            targetWinsToFetch = targetWins,
            sessionStartedTimestamp = currentTimeMs,
            sessionTimeoutRemainingMs = timeoutLimitMs,
            currentCalculatedStake = initialStake,
            currentStakeLevel = 1,
            consecutiveWinsInSession = 0,
            consecutiveLossesInSession = 0
        )

        publishStakeInstruction(
            stake = initialStake,
            mode = mode,
            addAmount = addAmt,
            targetMultiplier = effectiveTargetMult,
            stakeLevel = 1,
            isSessionActive = true,
            reason = "$strategyName Triggered"
        )
        triggerAction(
            AutomationActionEvent(
                actionType = ActionType.BET,
                amountOrMultiplier = initialStake,
                reason = "$strategyName matched -> Fetch $targetWins wins",
                timestamp = currentTimeMs
            )
        )
        safeLogI("External pattern strategy '$strategyName' activated betting session (stake=$initialStake)")
        return true
    }

    /**
     * Evaluates completed round multipliers against all rules while strictly enforcing
     * exclusive non-overlapping strategy lock and 3-minute session timeout guard.
     */
    fun processCompletedRound(
        multiplier: Double,
        config: StrategyConfig = _boundConfig.value,
        currentTimeMs: Long = System.currentTimeMillis()
    ): StrategyTrigger {
        totalRounds++

        // Clean up high multiplier loss timestamps older than 1 hour (3600000 ms)
        highMultiplierLossTimestamps.removeAll { currentTimeMs - it > 3_600_000L }

        // Enforce 3-minute timeout guard & check elapsed wait timers first
        val timerTrigger = checkPendingTimers(config, currentTimeMs)
        val timeoutLimitMs = config.sessionTimeoutMinutes.coerceAtLeast(1) * 60_000L

        // Exclusive Lock Check: while a betting session is active, do NOT start or overlap any other strategy
        if (state.isBettingSessionActive) {
            state = state.copy(
                highMultiplierLossesInHour = highMultiplierLossTimestamps.size,
                isNearHourEndRuleActive = shouldContinueBettingBasedOnHourRule(currentTimeMs)
            )
            return StrategyTrigger.None
        }

        var consecutiveLow = state.consecutiveLowCount
        var consecutiveObservedLosses = state.consecutiveObservedLosses
        var postHighWait = state.postHighWaitCount
        var isWaiting = state.isWaitingForPostHigh
        var lastHigh = state.lastHighMultiplier

        // Track consecutive observed losses (< targetMultiplier) while unlocked
        if (multiplier < config.targetMultiplier) {
            consecutiveObservedLosses += 1
        } else {
            consecutiveObservedLosses = 0
        }

        // Strategy 1 Evaluation: Count consecutive rounds < strategy1LowThreshold
        if (config.isStrategy1Enabled && multiplier < config.strategy1LowThreshold) {
            consecutiveLow += 1
        } else {
            consecutiveLow = 0
        }

        // Strategy 2 Evaluation: Detect > strategy2HighThreshold and count wait rounds
        if (config.isStrategy2Enabled) {
            if (multiplier > config.strategy2HighThreshold && !isWaiting) {
                isWaiting = true
                postHighWait = 0
                lastHigh = multiplier
            } else if (isWaiting) {
                postHighWait += 1
            }
        }

        val allRules = config.getRuleConfigs()
        var pendingWaitRuleId = state.pendingWaitRuleId
        var pendingWaitRuleLabel = state.pendingWaitRuleLabel
        var waitUntilTimestamp = state.waitUntilTimestamp

        // Evaluate Multiplier-Range Waiting Rules using editable min/max/waitMinutes
        var rangeTrigger: StrategyTrigger? = null
        if (pendingWaitRuleId == null) {
            val matchedRangeRule = allRules.filter {
                it.isEnabled &&
                    it.ruleId.category == com.example.data.model.StrategyCategory.MULTIPLIER_WAIT &&
                    it.waitDurationMs > 0L &&
                    multiplier > it.minMultiplier &&
                    multiplier < it.maxMultiplier
            }.maxByOrNull { it.minMultiplier }

            if (matchedRangeRule != null) {
                pendingWaitRuleId = matchedRangeRule.ruleId
                pendingWaitRuleLabel = matchedRangeRule.dynamicTitle()
                waitUntilTimestamp = currentTimeMs + matchedRangeRule.waitDurationMs
                rangeTrigger = StrategyTrigger.RangeWaitStarted(
                    ruleConfig = matchedRangeRule,
                    observedMultiplier = multiplier,
                    waitUntilTimestamp = waitUntilTimestamp
                )
                safeLogI("Range rule triggered: ${matchedRangeRule.dynamicTitle()} at ${multiplier}x")
            }
        }

        // Evaluate Loss Recovery Rules using editable requiredLosses (only if unlocked & no wait timer pending)
        var triggerFired: StrategyTrigger = if (timerTrigger is StrategyTrigger.WaitTimerCompleted) {
            timerTrigger
        } else {
            StrategyTrigger.None
        }

        var activatedRule: StrategyRuleConfig? = null
        if (triggerFired is StrategyTrigger.None && pendingWaitRuleId == null) {
            val matchingLossRule = allRules
                .filter {
                    it.isEnabled &&
                        it.ruleId.category != com.example.data.model.StrategyCategory.MULTIPLIER_WAIT &&
                        it.requiredLosses > 0 &&
                        consecutiveObservedLosses >= it.requiredLosses
                }
                .minByOrNull { it.requiredLosses }

            if (matchingLossRule != null) {
                activatedRule = matchingLossRule
                triggerFired = StrategyTrigger.LossRecoveryFired(
                    ruleConfig = matchingLossRule,
                    observedLosses = consecutiveObservedLosses
                )
            } else if (config.isStrategy1Enabled && consecutiveLow >= config.strategy1RequiredLowRounds) {
                triggerFired = StrategyTrigger.Strategy1Fired(consecutiveLow)
            } else if (config.isStrategy2Enabled && isWaiting && postHighWait >= config.strategy2WaitRounds) {
                triggerFired = StrategyTrigger.Strategy2Fired(postHighWait, lastHigh)
            }
        }

        val hourRuleActive = shouldContinueBettingBasedOnHourRule(currentTimeMs)
        val sessionActive = state.isBettingSessionActive ||
            (triggerFired is StrategyTrigger.LossRecoveryFired ||
                triggerFired is StrategyTrigger.WaitTimerCompleted ||
                triggerFired is StrategyTrigger.Strategy1Fired ||
                triggerFired is StrategyTrigger.Strategy2Fired ||
                hourRuleActive)

        val initialStake = if (!state.isBettingSessionActive && sessionActive) {
            config.baseStake
        } else {
            state.currentCalculatedStake
        }

        val resolvedActiveRuleId = when {
            activatedRule != null -> activatedRule.ruleId
            triggerFired is StrategyTrigger.WaitTimerCompleted -> triggerFired.ruleConfig.ruleId
            state.isBettingSessionActive -> state.activeRuleId
            else -> null
        }

        val resolvedRuleConfig = resolvedActiveRuleId?.let { config.getRuleConfig(it) }
        val resolvedRuleLabel = when {
            resolvedRuleConfig != null -> resolvedRuleConfig.dynamicTitle()
            triggerFired is StrategyTrigger.Strategy1Fired -> "Strategy 1 (<${config.strategy1LowThreshold}x)"
            triggerFired is StrategyTrigger.Strategy2Fired -> "Strategy 2 (>${config.strategy2HighThreshold}x)"
            else -> state.activeRuleLabel
        }

        val sessionStarted = when {
            !state.isBettingSessionActive && sessionActive -> currentTimeMs
            sessionActive -> state.sessionStartedTimestamp
            else -> 0L
        }

        val (effectiveMode, effectiveAddAmount) = resolveEffectiveProgression(config)
        state = state.copy(
            consecutiveLowCount = consecutiveLow,
            consecutiveObservedLosses = consecutiveObservedLosses,
            postHighWaitCount = postHighWait,
            lastHighMultiplier = lastHigh,
            isWaitingForPostHigh = if (isWaiting && postHighWait >= config.strategy2WaitRounds) false else isWaiting,
            activeRuleId = resolvedActiveRuleId,
            activeRuleLabel = resolvedRuleLabel,
            activeProgressionMode = effectiveMode,
            activeCustomAddAmount = effectiveAddAmount,
            targetWinsToFetch = resolvedRuleConfig?.targetWinsToFetch ?: config.stopAfterConsecutiveWins,
            pendingWaitRuleId = pendingWaitRuleId,
            pendingWaitRuleLabel = pendingWaitRuleLabel,
            waitUntilTimestamp = waitUntilTimestamp,
            activeWaitRemainingMs = if (pendingWaitRuleId != null) (waitUntilTimestamp - currentTimeMs).coerceAtLeast(0L) else 0L,
            activeTrigger = when {
                triggerFired !is StrategyTrigger.None -> triggerFired
                rangeTrigger != null -> rangeTrigger
                timerTrigger is StrategyTrigger.SessionTimedOut -> timerTrigger
                else -> state.activeTrigger
            },
            isBettingSessionActive = sessionActive,
            sessionStartedTimestamp = sessionStarted,
            sessionTimeoutRemainingMs = if (sessionActive) timeoutLimitMs else 0L,
            currentCalculatedStake = initialStake,
            highMultiplierLossesInHour = highMultiplierLossTimestamps.size,
            isNearHourEndRuleActive = hourRuleActive
        )

        when (val fired = triggerFired) {
            is StrategyTrigger.LossRecoveryFired -> {
                publishStakeInstruction(
                    stake = initialStake,
                    mode = effectiveMode,
                    addAmount = effectiveAddAmount,
                    targetMultiplier = config.targetMultiplier,
                    stakeLevel = state.currentStakeLevel,
                    isSessionActive = true,
                    reason = "${fired.ruleConfig.dynamicTitle()} Triggered"
                )
                triggerAction(
                    AutomationActionEvent(
                        actionType = ActionType.BET,
                        amountOrMultiplier = initialStake,
                        reason = "${fired.ruleConfig.dynamicTitle()} (${fired.observedLosses} losses) -> Fetch ${fired.ruleConfig.targetWinsToFetch} wins",
                        timestamp = currentTimeMs
                    )
                )
            }
            is StrategyTrigger.Strategy1Fired -> {
                publishStakeInstruction(
                    stake = initialStake,
                    mode = state.activeProgressionMode,
                    addAmount = state.activeCustomAddAmount,
                    targetMultiplier = config.targetMultiplier,
                    stakeLevel = state.currentStakeLevel,
                    isSessionActive = true,
                    reason = "Strategy 1 Triggered"
                )
                triggerAction(
                    AutomationActionEvent(
                        actionType = ActionType.BET,
                        amountOrMultiplier = initialStake,
                        reason = "Strategy 1: <${config.strategy1LowThreshold}x streak ($consecutiveLow rounds)",
                        timestamp = currentTimeMs
                    )
                )
            }
            is StrategyTrigger.Strategy2Fired -> {
                publishStakeInstruction(
                    stake = initialStake,
                    mode = state.activeProgressionMode,
                    addAmount = state.activeCustomAddAmount,
                    targetMultiplier = config.targetMultiplier,
                    stakeLevel = state.currentStakeLevel,
                    isSessionActive = true,
                    reason = "Strategy 2 Triggered"
                )
                triggerAction(
                    AutomationActionEvent(
                        actionType = ActionType.BET,
                        amountOrMultiplier = initialStake,
                        reason = "Strategy 2: >${config.strategy2HighThreshold}x (${lastHigh}x) + ${config.strategy2WaitRounds} rounds wait",
                        timestamp = currentTimeMs
                    )
                )
            }
            else -> Unit
        }

        return when {
            triggerFired !is StrategyTrigger.None -> triggerFired
            rangeTrigger != null -> rangeTrigger
            timerTrigger is StrategyTrigger.SessionTimedOut -> timerTrigger
            else -> StrategyTrigger.None
        }
    }

    /**
     * Process betting outcome for an active automated bet.
     * Enforces per-strategy stake progression (Option A: Add Amount vs Option B: Double Amount),
     * target wins reset (e.g., fetch 2 wins then unlock/reset), and >13.00x loss rule,
     * publishing the next StakeInstruction ready for AccessibilityController.
     */
    fun processBetOutcome(
        isWin: Boolean,
        multiplier: Double,
        config: StrategyConfig = _boundConfig.value,
        currentTimeMs: Long = System.currentTimeMillis()
    ): StrategyState {
        val activeRuleConfig = state.activeRuleId?.let { config.getRuleConfig(it) }
        val targetWins = activeRuleConfig?.targetWinsToFetch ?: config.stopAfterConsecutiveWins
        val (activeMode, activeAddAmount) = resolveEffectiveProgression(config)

        if (isWin) {
            totalWins++
            val wins = state.consecutiveWinsInSession + 1
            safeLogI("Bet Won! Verified wins in recovery session: $wins / $targetWins")
            triggerAction(
                AutomationActionEvent(
                    actionType = ActionType.CASH_OUT,
                    amountOrMultiplier = multiplier,
                    reason = "Auto Cashout Won ($wins/$targetWins) @ ${String.format(Locale.US, "%.2f", multiplier)}x",
                    timestamp = currentTimeMs
                )
            )

            val resetBase = roundStake(config.baseStake)
            val shouldResetOnWin = (config.isMartingaleEnabled && activeMode != StakeProgressionMode.DISABLED) ||
                (wins >= targetWins && !shouldContinueBettingBasedOnHourRule(currentTimeMs))
            if (shouldResetOnWin) {
                state = state.copy(
                    isBettingSessionActive = false,
                    sessionStartedTimestamp = 0L,
                    sessionTimeoutRemainingMs = 0L,
                    activeTrigger = StrategyTrigger.None,
                    activeRuleId = null,
                    activeRuleLabel = null,
                    activeProgressionMode = activeMode,
                    activeCustomAddAmount = activeAddAmount,
                    currentStakeLevel = 1,
                    consecutiveWinsInSession = 0,
                    consecutiveLossesInSession = 0,
                    consecutiveObservedLosses = 0,
                    currentCalculatedStake = resetBase,
                    consecutiveLowCount = 0,
                    postHighWaitCount = 0
                )
                publishStakeInstruction(
                    stake = resetBase,
                    mode = activeMode,
                    addAmount = activeAddAmount,
                    targetMultiplier = config.targetMultiplier,
                    stakeLevel = 1,
                    isSessionActive = false,
                    reason = if (config.isMartingaleEnabled && activeMode != StakeProgressionMode.DISABLED) {
                        "Martingale Win -> Reset to Base Stake"
                    } else {
                        "Target $targetWins Wins Fetched -> Reset to Base Stake"
                    }
                )
            } else {
                state = state.copy(
                    consecutiveWinsInSession = wins,
                    consecutiveLossesInSession = 0
                )
                publishStakeInstruction(
                    stake = state.currentCalculatedStake,
                    mode = activeMode,
                    addAmount = activeAddAmount,
                    targetMultiplier = config.targetMultiplier,
                    stakeLevel = state.currentStakeLevel,
                    isSessionActive = true,
                    reason = "Win $wins/$targetWins Verified -> Continue Session"
                )
            }
        } else {
            totalLosses++
            val losses = state.consecutiveLossesInSession + 1

            // Strictly obey master progression setting (DISABLED stays flat regardless of Martingale or rule mode)
            val nextStake = calculateNextStakeAfterLoss(
                currentStake = state.currentCalculatedStake,
                progressionMode = activeMode,
                customAddAmount = activeAddAmount,
                lossMultiplier = multiplier,
                isHighMultiplierLossRuleEnabled = false
            )

            safeLogI("Bet Lost. Session losses: $losses, masterMode: $activeMode, martingale: ${config.isMartingaleEnabled}, next stake: $nextStake")

            val hourRuleActive = shouldContinueBettingBasedOnHourRule(currentTimeMs)
            val newLevel = state.currentStakeLevel + 1

            state = state.copy(
                isBettingSessionActive = true,
                activeProgressionMode = activeMode,
                activeCustomAddAmount = activeAddAmount,
                currentStakeLevel = newLevel,
                consecutiveWinsInSession = 0,
                consecutiveLossesInSession = losses,
                currentCalculatedStake = nextStake,
                highMultiplierLossesInHour = highMultiplierLossTimestamps.size,
                isNearHourEndRuleActive = hourRuleActive
            )

            publishStakeInstruction(
                stake = nextStake,
                mode = activeMode,
                addAmount = activeAddAmount,
                targetMultiplier = config.targetMultiplier,
                stakeLevel = newLevel,
                isSessionActive = true,
                reason = "Loss #$losses (${activeMode.label}) -> Next Stake ${formatStakeAmount(nextStake)}"
            )
        }

        return state
    }

    /**
     * Checks if 3 high-multiplier losses occurred in the current hour and if it is
     * within 10 minutes of the hour ending (minute >= 50).
     */
    private fun shouldContinueBettingBasedOnHourRule(currentTimeMs: Long = System.currentTimeMillis()): Boolean {
        if (highMultiplierLossTimestamps.size < 3) return false
        val calendar = Calendar.getInstance().apply { timeInMillis = currentTimeMs }
        val minute = calendar.get(Calendar.MINUTE)
        return minute >= 50
    }
}

    val matchedCount: Int = 0,
    val requiredCount: Int = 1,
    val isApproaching: Boolean = false,
    val isReached: Boolean = false,
    val streakStartKey: String? = null,
    val streakStartTimestampMs: Long = 0L,
    val lastEvaluatedRecordKey: String? = null,
    val lastEvaluatedMultiplier: Double? = null
)

data class PatternEvaluationBatch(
    val updatedStrategies: List<CustomPatternStrategy>,
    val events: List<PatternAlertEvent>
)

/**
 * Chronological, non-predictive pattern detection engine for Section 15.
 * - Evaluates user-defined CustomPatternStrategy conditions in chronological order.
 * - Prevents double-counting via per-strategy lastEvaluatedRecordKey tracking.
 * - Resets consecutive streaks immediately when a breaking round occurs and emits
 *   STRATEGY_INVALIDATED when an approaching or active occurrence is broken.
 * - Uses strictly neutral factual status language (never predictive claims).
 */
class PatternDetectionEngine {

    companion object {
        fun formatMultiplier(value: Double): String = String.format(Locale.US, "%.2fx", value)
    }

    /**
     * Evaluates a single condition across a chronologically sorted list (`oldest -> newest`).
     * Uses records filtered to `condition.timeWindowMinutes` (relative to `nowMs`) when specified.
     */
    fun evaluateCondition(
        condition: PatternCondition,
        chronologicalRecords: List<MultiplierRecord>,
        nowMs: Long = System.currentTimeMillis()
    ): PatternConditionProgress {
        val scopedRecords = if (condition.timeWindowMinutes != null && condition.timeWindowMinutes > 0) {
            val cutoff = nowMs - (condition.timeWindowMinutes * 60_000L)
            chronologicalRecords.filter { it.crashTimestamp >= cutoff }
        } else {
            chronologicalRecords
        }

        if (scopedRecords.isEmpty()) {
            val req = if (condition.sequenceMultipliers.isNotEmpty()) {
                condition.sequenceMultipliers.size
            } else {
                condition.consecutiveRounds.coerceAtLeast(1)
            }
            return PatternConditionProgress(matchedCount = 0, requiredCount = req)
        }

        // Mode 1: Exact sequence pattern check against trailing rounds
        if (condition.sequenceMultipliers.isNotEmpty()) {
            val seq = condition.sequenceMultipliers
            val req = seq.size
            var longestTailMatch = 0
            var startKey: String? = null
            var startTs = 0L

            for (startIdx in scopedRecords.indices) {
                var matched = 0
                for (offset in seq.indices) {
                    val recIdx = startIdx + offset
                    if (recIdx >= scopedRecords.size) break
                    if (abs(scopedRecords[recIdx].multiplier - seq[offset]) <= 0.01) {
                        matched++
                    } else {
                        break
                    }
                }
                if (startIdx + matched == scopedRecords.size && matched > 0) {
                    longestTailMatch = matched
                    startKey = scopedRecords[startIdx].dedupKey
                    startTs = scopedRecords[startIdx].crashTimestamp
                    break
                }
            }

            val reached = longestTailMatch >= req
            val approaching = !reached && req >= 2 && longestTailMatch == req - 1
            val lastRec = scopedRecords.last()
            return PatternConditionProgress(
                matchedCount = longestTailMatch,
                requiredCount = req,
                isApproaching = approaching,
                isReached = reached,
                streakStartKey = startKey,
                streakStartTimestampMs = startTs,
                lastEvaluatedRecordKey = lastRec.dedupKey,
                lastEvaluatedMultiplier = lastRec.multiplier
            )
        }

        // Mode 2: Consecutive rounds meeting threshold comparison
        val required = condition.consecutiveRounds.coerceAtLeast(1)
        var streak = 0
        var streakStartKey: String? = null
        var streakStartTs = 0L

        for (rec in scopedRecords) {
            if (condition.comparator.matches(rec.multiplier, condition.threshold, condition.upperThreshold)) {
                if (streak == 0) {
                    streakStartKey = rec.dedupKey
                    streakStartTs = rec.crashTimestamp
                }
                streak++
            } else {
                streak = 0
                streakStartKey = null
                streakStartTs = 0L
            }
        }

        val reached = streak >= required
        val approaching = !reached && required >= 2 && streak == required - 1
        val lastRec = scopedRecords.last()
        return PatternConditionProgress(
            matchedCount = streak,
            requiredCount = required,
            isApproaching = approaching,
            isReached = reached,
            streakStartKey = streakStartKey,
            streakStartTimestampMs = streakStartTs,
            lastEvaluatedRecordKey = lastRec.dedupKey,
            lastEvaluatedMultiplier = lastRec.multiplier
        )
    }

    /**
     * Evaluates all enabled [CustomPatternStrategy] rules against the current history records.
     * Processes only records strictly newer than each strategy's `lastEvaluatedRecordKey` step-by-step
     * so every intermediate approaching -> reached -> invalidated transition is captured accurately
     * without double-counting any round.
     */
    fun evaluateAllStrategies(
        strategies: List<CustomPatternStrategy>,
        records: List<MultiplierRecord>,
        nowMs: Long = System.currentTimeMillis()
    ): PatternEvaluationBatch {
        if (strategies.isEmpty() || records.isEmpty()) {
            return PatternEvaluationBatch(updatedStrategies = strategies, events = emptyList())
        }

        val chronological = records.sortedWith(
            compareBy<MultiplierRecord>(
                { it.crashTimestamp },
                { -it.sequenceIndex },
                { it.id }
            )
        )

        val updatedStrategies = mutableListOf<CustomPatternStrategy>()
        val emittedEvents = mutableListOf<PatternAlertEvent>()

        for (strategy in strategies) {
            if (!strategy.isEnabled) {
                updatedStrategies.add(strategy)
                continue
            }

            val conditions = strategy.getConditions()
            if (conditions.isEmpty()) {
                updatedStrategies.add(strategy)
                continue
            }

            val lastIdx = if (strategy.lastEvaluatedRecordKey != null) {
                chronological.indexOfLast { it.dedupKey == strategy.lastEvaluatedRecordKey }
            } else {
                -1
            }

            // If no new records exist after lastEvaluatedRecordKey, check only optional cooldown repeat
            if (lastIdx >= chronological.lastIndex) {
                var currentStrategy = strategy
                if (strategy.currentlyReached &&
                    strategy.repeatPolicy == RepeatNotificationPolicy.COOLDOWN_MINUTES &&
                    strategy.repeatCooldownMinutes > 0 &&
                    strategy.lastNotifiedAtMs > 0L &&
                    (nowMs - strategy.lastNotifiedAtMs) >= strategy.repeatCooldownMinutes * 60_000L
                ) {
                    val latestRec = chronological.last()
                    val occKey = strategy.lastTriggeredOccurrenceKey ?: "${strategy.id}_reach_${latestRec.dedupKey}"
                    emittedEvents.add(
                        buildEvent(
                            strategy = currentStrategy,
                            type = StrategyNotificationType.STRATEGY_REACHED,
                            triggerRec = latestRec,
                            occurrenceKey = occKey,
                            progressSummary = strategy.summaryDescription(),
                            nowMs = nowMs
                        )
                    )
                    currentStrategy = currentStrategy.copy(lastNotifiedAtMs = nowMs)
                }
                updatedStrategies.add(currentStrategy)
                continue
            }

            // Replay step-by-step through newly unseen records so rapid batches (e.g. 7 then 8th) never miss milestones
            var workingStrategy = strategy
            val startIndex = (lastIdx + 1).coerceAtLeast(0)
            for (endIdx in startIndex until chronological.size) {
                val prefixSlice = chronological.subList(0, endIdx + 1)
                val latestRec = chronological[endIdx]
                val stepResult = evaluateSingleSnapshotStep(
                    strategy = workingStrategy,
                    conditions = conditions,
                    chronologicalPrefix = prefixSlice,
                    latestRecord = latestRec,
                    nowMs = nowMs
                )
                workingStrategy = stepResult.first
                emittedEvents.addAll(stepResult.second)
            }
            updatedStrategies.add(workingStrategy)
        }

        return PatternEvaluationBatch(
            updatedStrategies = updatedStrategies,
            events = emittedEvents
        )
    }

    private fun evaluateSingleSnapshotStep(
        strategy: CustomPatternStrategy,
        conditions: List<PatternCondition>,
        chronologicalPrefix: List<MultiplierRecord>,
        latestRecord: MultiplierRecord,
        nowMs: Long
    ): Pair<CustomPatternStrategy, List<PatternAlertEvent>> {
        val progresses = conditions.map { evaluateCondition(it, chronologicalPrefix, nowMs) }
        val events = mutableListOf<PatternAlertEvent>()

        val anyBroken = progresses.any { !it.isReached && !it.isApproaching }
        val allReached = progresses.all { it.isReached }
        val anyReached = progresses.any { it.isReached }
        val isReachedNow = when (strategy.logicOperator) {
            PatternLogicOperator.AND -> allReached
            PatternLogicOperator.OR -> anyReached
        }

        val isApproachingNow = when (strategy.logicOperator) {
            PatternLogicOperator.AND -> !isReachedNow &&
                progresses.any { it.isApproaching } &&
                progresses.all { it.isApproaching || it.isReached }
            PatternLogicOperator.OR -> !isReachedNow && progresses.any { it.isApproaching }
        }

        val anchorKey = progresses.firstOrNull { it.streakStartKey != null }?.streakStartKey ?: latestRecord.dedupKey
        val waitingSatisfied = if (strategy.waitingPeriodSeconds <= 0) {
            true
        } else {
            val triggerWindowStart = progresses.maxOfOrNull { it.streakStartTimestampMs } ?: latestRecord.crashTimestamp
            (nowMs - triggerWindowStart) >= strategy.waitingPeriodSeconds * 1000L
        }

        var nextStrategy = strategy.copy(lastEvaluatedRecordKey = latestRecord.dedupKey)

        when {
            isReachedNow && waitingSatisfied -> {
                val occKey = "${strategy.id}_reach_$anchorKey"
                val shouldNotifyNew = strategy.lastTriggeredOccurrenceKey != occKey
                val shouldNotifyRepeat = !shouldNotifyNew &&
                    strategy.repeatPolicy == RepeatNotificationPolicy.COOLDOWN_MINUTES &&
                    strategy.repeatCooldownMinutes > 0 &&
                    (nowMs - strategy.lastNotifiedAtMs) >= strategy.repeatCooldownMinutes * 60_000L

                if (shouldNotifyNew || shouldNotifyRepeat) {
                    val primaryProg = progresses.firstOrNull { it.isReached } ?: progresses.first()
                    val summary = if (primaryProg.requiredCount > 1) {
                        "${primaryProg.matchedCount} of ${primaryProg.requiredCount} consecutive rounds (${strategy.summaryDescription()})"
                    } else {
                        strategy.summaryDescription()
                    }
                    events.add(
                        buildEvent(
                            strategy = nextStrategy,
                            type = StrategyNotificationType.STRATEGY_REACHED,
                            triggerRec = latestRecord,
                            occurrenceKey = occKey,
                            progressSummary = summary,
                            nowMs = nowMs
                        )
                    )
                    nextStrategy = nextStrategy.copy(
                        lastTriggeredOccurrenceKey = occKey,
                        lastNotifiedAtMs = nowMs
                    )
                }
                nextStrategy = nextStrategy.copy(
                    currentlyApproaching = false,
                    currentlyReached = true
                )
            }

            isApproachingNow && strategy.approachingEnabled -> {
                val occKey = "${strategy.id}_appr_$anchorKey"
                if (strategy.lastApproachedOccurrenceKey != occKey) {
                    val primaryProg = progresses.firstOrNull { it.isApproaching } ?: progresses.first()
                    val summary = "${primaryProg.matchedCount} of ${primaryProg.requiredCount} (${strategy.summaryDescription()})"
                    events.add(
                        buildEvent(
                            strategy = nextStrategy,
                            type = StrategyNotificationType.STRATEGY_APPROACHING,
                            triggerRec = latestRecord,
                            occurrenceKey = occKey,
                            progressSummary = summary,
                            nowMs = nowMs
                        )
                    )
                    nextStrategy = nextStrategy.copy(
                        lastApproachedOccurrenceKey = occKey,
                        lastNotifiedAtMs = nowMs
                    )
                }
                nextStrategy = nextStrategy.copy(
                    currentlyApproaching = true,
                    currentlyReached = false
                )
            }

            else -> {
                val wasActiveOrApproaching = strategy.currentlyApproaching || strategy.currentlyReached
                if (wasActiveOrApproaching && (anyBroken || !isReachedNow)) {
                    val occKey = "${strategy.id}_inv_${latestRecord.dedupKey}"
                    events.add(
                        buildEvent(
                            strategy = nextStrategy,
                            type = StrategyNotificationType.STRATEGY_INVALIDATED,
                            triggerRec = latestRecord,
                            occurrenceKey = occKey,
                            progressSummary = "Condition reset after ${formatMultiplier(latestRecord.multiplier)}",
                            nowMs = nowMs
                        )
                    )
                    nextStrategy = nextStrategy.copy(lastNotifiedAtMs = nowMs)
                }
                nextStrategy = nextStrategy.copy(
                    currentlyApproaching = false,
                    currentlyReached = false
                )
            }
        }

        return nextStrategy to events
    }

    private fun buildEvent(
        strategy: CustomPatternStrategy,
        type: StrategyNotificationType,
        triggerRec: MultiplierRecord,
        occurrenceKey: String,
        progressSummary: String,
        nowMs: Long
    ): PatternAlertEvent {
        val customNote = strategy.customMessage.trim().takeIf { it.isNotEmpty() }?.let { " $it" }.orEmpty()
        val (title, message) = when (type) {
            StrategyNotificationType.STRATEGY_APPROACHING -> {
                "Pattern Approaching: ${strategy.name}" to
                    "PATTERN APPROACHING: $progressSummary. Latest round: ${formatMultiplier(triggerRec.multiplier)}. Current status: Monitoring.$customNote"
            }
            StrategyNotificationType.STRATEGY_REACHED -> {
                "Pattern Detected: ${strategy.name}" to
                    "PATTERN DETECTED: $progressSummary. Latest round: ${formatMultiplier(triggerRec.multiplier)}. Open the game and review the current betting opportunity.$customNote"
            }
            StrategyNotificationType.STRATEGY_INVALIDATED -> {
                "Pattern Reset: ${strategy.name}" to
                    "PATTERN INVALIDATED: ${strategy.name} — $progressSummary. Waiting for next sequence."
            }
            StrategyNotificationType.HISTORY_UNAVAILABLE -> {
                "History Unavailable: ${strategy.name}" to
                    "HISTORY UNAVAILABLE: Game ribbon history could not be read; pattern monitoring is paused until history resumes."
            }
            StrategyNotificationType.STRATEGY_PAUSED -> {
                "Strategy Paused: ${strategy.name}" to
                    "STRATEGY PAUSED: ${strategy.name} monitoring paused."
            }
            else -> {
                "Predict Manager Update: ${strategy.name}" to
                    "${type.displayLabel}: $progressSummary"
            }
        }

        return PatternAlertEvent(
            strategy = strategy,
            notificationType = type,
            title = title,
            message = message,
            triggeringRecordKey = triggerRec.dedupKey,
            triggeringMultiplier = triggerRec.multiplier,
            occurrenceKey = occurrenceKey,
            timestampMs = nowMs
        )
    }

    /**
     * Evaluates Strategy A (Consecutive Lows), Strategy B (Big Multiplier Intervals),
     * Strategy C (Color Patterns), and Custom strategies against completed CrashRecord entries.
     */
    fun evaluateStrategiesOnCrashRecords(
        strategies: List<CustomPatternStrategy>,
        crashRecords: List<CrashRecord>,
        lowThreshold: Double = 2.00,
        bigThreshold: Double = 10.00,
        nowMs: Long = System.currentTimeMillis()
    ): CrashPatternEvaluationBatch {
        if (strategies.isEmpty() || crashRecords.isEmpty()) {
            return CrashPatternEvaluationBatch(updatedStrategies = strategies, events = emptyList(), newOccurrences = emptyList())
        }

        val updatedStrategies = mutableListOf<CustomPatternStrategy>()
        val alertEvents = mutableListOf<PatternAlertEvent>()
        val occurrences = mutableListOf<com.example.data.model.PatternOccurrenceRecord>()

        val chronological = crashRecords.sortedWith(compareBy({ it.crashTimestamp }, { it.id }))

        for (strategy in strategies) {
            if (!strategy.isEnabled) {
                updatedStrategies.add(strategy)
                continue
            }

            val gameScoped = if (strategy.targetGameId == "ALL") {
                val activePartitionGameId = chronological.last().gameId
                chronological.filter { it.gameId == activePartitionGameId }
            } else {
                chronological.filter { it.gameId == strategy.targetGameId }
            }
            if (gameScoped.isEmpty()) {
                updatedStrategies.add(strategy)
                continue
            }

            val latest = gameScoped.last()
            if (strategy.lastEvaluatedRecordKey == latest.dedupKey) {
                updatedStrategies.add(strategy)
                continue
            }

            if (strategy.multiplierRuleSubtype == com.example.data.model.MultiplierRuleSubtype.ABOVE_THEN_BELOW) {
                val aboveN = strategy.aboveConsecutiveCount.coerceAtLeast(1)
                val belowM = strategy.belowConsecutiveCount.coerceAtLeast(1)
                val totalNeeded = aboveN + belowM
                var nextStrat = strategy.copy(lastEvaluatedRecordKey = latest.dedupKey)

                val exactTailWindow = if (gameScoped.size >= totalNeeded) gameScoped.takeLast(totalNeeded) else null
                val exactReached = exactTailWindow != null && run {
                    val aboveSlice = exactTailWindow.take(aboveN)
                    val belowSlice = exactTailWindow.takeLast(belowM)
                    val aboveOk = aboveSlice.all { r ->
                        if (strategy.inclusiveBounds) r.multiplier >= strategy.aboveThresholdX else r.multiplier > strategy.aboveThresholdX
                    }
                    val belowOk = belowSlice.all { r ->
                        if (strategy.inclusiveBounds) r.multiplier <= strategy.belowThresholdY else r.multiplier < strategy.belowThresholdY
                    }
                    aboveOk && belowOk
                }

                if (exactReached && exactTailWindow != null) {
                    val anchor = exactTailWindow.first()
                    val occKey = "aboveBelow_${strategy.id}_${latest.gameId}_${anchor.dedupKey}"
                    if (strategy.lastTriggeredOccurrenceKey != occKey) {
                        val opAbove = if (strategy.inclusiveBounds) ">=" else ">"
                        val opBelow = if (strategy.inclusiveBounds) "<=" else "<"
                        val desc = "${aboveN}x $opAbove ${formatMultiplier(strategy.aboveThresholdX)} then ${belowM}x $opBelow ${formatMultiplier(strategy.belowThresholdY)}"
                        val hourStart = com.example.data.model.CalendarHourUtils.hourStartFor(latest.crashTimestamp)
                        occurrences.add(
                            com.example.data.model.PatternOccurrenceRecord(
                                occurrenceKey = occKey,
                                strategyId = strategy.id,
                                strategyName = strategy.name,
                                gameId = latest.gameId,
                                gameName = latest.gameName,
                                patternType = "ABOVE_THEN_BELOW",
                                triggerTimestampMs = latest.crashTimestamp,
                                hourStartTimestamp = hourStart,
                                detailsSummary = desc,
                                firstMatchedTimestampMs = anchor.crashTimestamp
                            )
                        )
                        alertEvents.add(
                            PatternAlertEvent(
                                strategy = nextStrat,
                                notificationType = StrategyNotificationType.STRATEGY_REACHED,
                                title = "${latest.gameName}: ${strategy.name}",
                                message = "PATTERN DETECTED: $desc (latest ${formatMultiplier(latest.multiplier)}).",
                                triggeringRecordKey = latest.dedupKey,
                                triggeringMultiplier = latest.multiplier,
                                occurrenceKey = occKey,
                                timestampMs = nowMs
                            )
                        )
                        nextStrat = nextStrat.copy(
                            lastTriggeredOccurrenceKey = occKey,
                            lastNotifiedAtMs = nowMs,
                            currentlyReached = true,
                            currentlyApproaching = false
                        )
                    }
                } else {
                    val approxNeeded = aboveN + (belowM - 1)
                    val isApprox = strategy.approachingEnabled && belowM >= 2 && gameScoped.size >= approxNeeded && run {
                        val sub = gameScoped.takeLast(approxNeeded)
                        val aOk = sub.take(aboveN).all { r ->
                            if (strategy.inclusiveBounds) r.multiplier >= strategy.aboveThresholdX else r.multiplier > strategy.aboveThresholdX
                        }
                        val bOk = sub.takeLast(belowM - 1).all { r ->
                            if (strategy.inclusiveBounds) r.multiplier <= strategy.belowThresholdY else r.multiplier < strategy.belowThresholdY
                        }
                        aOk && bOk
                    }
                    if (isApprox) {
                        val anchor = gameScoped[gameScoped.size - approxNeeded]
                        val occKey = "aboveBelow_appr_${strategy.id}_${latest.gameId}_${anchor.dedupKey}"
                        if (strategy.lastApproachedOccurrenceKey != occKey) {
                            alertEvents.add(
                                PatternAlertEvent(
                                    strategy = nextStrat,
                                    notificationType = StrategyNotificationType.STRATEGY_APPROACHING,
                                    title = "Approaching: ${strategy.name}",
                                    message = "PATTERN APPROACHING: $aboveN above ${formatMultiplier(strategy.aboveThresholdX)} followed by ${belowM - 1}/$belowM below ${formatMultiplier(strategy.belowThresholdY)}.",
                                    triggeringRecordKey = latest.dedupKey,
                                    triggeringMultiplier = latest.multiplier,
                                    occurrenceKey = occKey,
                                    timestampMs = nowMs
                                )
                            )
                            nextStrat = nextStrat.copy(lastApproachedOccurrenceKey = occKey, lastNotifiedAtMs = nowMs)
                        }
                        nextStrat = nextStrat.copy(currentlyApproaching = true, currentlyReached = false)
                    } else {
                        nextStrat = nextStrat.copy(currentlyApproaching = false, currentlyReached = false)
                    }
                }
                updatedStrategies.add(nextStrat)
                continue
            }

            when (strategy.strategyMode) {
                com.example.data.model.StrategyMode.STRATEGY_A_CONSECUTIVE_LOWS -> {
                    val cond = strategy.getConditions().firstOrNull()
                    val th = cond?.threshold ?: lowThreshold
                    val requiredCount = (cond?.consecutiveRounds ?: 14).coerceAtLeast(1)
                    val windowCutoff = nowMs - (strategy.analysisWindowMinutes.coerceAtLeast(5) * 60_000L)
                    val windowed = gameScoped.filter { it.crashTimestamp >= windowCutoff }

                    var streak = 0
                    var startRec: CrashRecord? = null
                    for (r in windowed) {
                        val matched = if (cond != null) {
                            cond.comparator.matches(r.multiplier, cond.threshold, cond.upperThreshold)
                        } else if (strategy.inclusiveBounds) {
                            r.multiplier <= th
                        } else {
                            r.multiplier < th
                        }
                        if (matched) {
                            if (streak == 0) startRec = r
                            streak++
                        } else {
                            streak = 0
                            startRec = null
                        }
                    }

                    val reached = streak >= requiredCount
                    val approaching = !reached && requiredCount >= 2 && streak == requiredCount - 1
                    var nextStrat = strategy.copy(lastEvaluatedRecordKey = latest.dedupKey)

                    if (reached) {
                        val occKey = "stratA_${strategy.id}_${latest.gameId}_${startRec?.dedupKey ?: latest.dedupKey}"
                        if (strategy.lastTriggeredOccurrenceKey != occKey) {
                            val hourStart = com.example.data.model.CalendarHourUtils.hourStartFor(latest.crashTimestamp)
                            occurrences.add(
                                com.example.data.model.PatternOccurrenceRecord(
                                    occurrenceKey = occKey,
                                    strategyId = strategy.id,
                                    strategyName = strategy.name,
                                    gameId = latest.gameId,
                                    gameName = latest.gameName,
                                    patternType = strategy.strategyMode.name,
                                    triggerTimestampMs = latest.crashTimestamp,
                                    hourStartTimestamp = hourStart,
                                    detailsSummary = "$streak consecutive rounds < ${String.format(Locale.US, "%.2fx", th)}",
                                    firstMatchedTimestampMs = startRec?.crashTimestamp ?: latest.crashTimestamp
                                )
                            )
                            alertEvents.add(
                                PatternAlertEvent(
                                    strategy = nextStrat,
                                    notificationType = StrategyNotificationType.STRATEGY_REACHED,
                                    title = "${latest.gameName}: ${strategy.name}",
                                    message = "Observed $streak consecutive rounds below ${String.format(Locale.US, "%.2fx", th)} (latest: ${formatMultiplier(latest.multiplier)}).",
                                    triggeringRecordKey = latest.dedupKey,
                                    triggeringMultiplier = latest.multiplier,
                                    occurrenceKey = occKey,
                                    timestampMs = nowMs
                                )
                            )
                            nextStrat = nextStrat.copy(lastTriggeredOccurrenceKey = occKey, lastNotifiedAtMs = nowMs)
                        }
                        nextStrat = nextStrat.copy(currentlyApproaching = false, currentlyReached = true)
                    } else if (approaching && strategy.approachingEnabled) {
                        val occKey = "stratA_appr_${strategy.id}_${latest.gameId}_${startRec?.dedupKey ?: latest.dedupKey}"
                        if (strategy.lastApproachedOccurrenceKey != occKey) {
                            alertEvents.add(
                                PatternAlertEvent(
                                    strategy = nextStrat,
                                    notificationType = StrategyNotificationType.STRATEGY_APPROACHING,
                                    title = "Approaching: ${strategy.name}",
                                    message = "$streak of $requiredCount consecutive rounds below ${String.format(Locale.US, "%.2fx", th)} in ${latest.gameName}.",
                                    triggeringRecordKey = latest.dedupKey,
                                    triggeringMultiplier = latest.multiplier,
                                    occurrenceKey = occKey,
                                    timestampMs = nowMs
                                )
                            )
                            nextStrat = nextStrat.copy(lastApproachedOccurrenceKey = occKey, lastNotifiedAtMs = nowMs)
                        }
                        nextStrat = nextStrat.copy(currentlyApproaching = true, currentlyReached = false)
                    } else {
                        nextStrat = nextStrat.copy(currentlyApproaching = false, currentlyReached = false)
                    }
                    updatedStrategies.add(nextStrat)
                }

                com.example.data.model.StrategyMode.STRATEGY_B_BIG_INTERVALS -> {
                    var nextStrat = strategy.copy(lastEvaluatedRecordKey = latest.dedupKey)
                    val bigTh = if (strategy.bigMultiplierThreshold > 1.0) strategy.bigMultiplierThreshold else bigThreshold
                    val bigs = gameScoped.filter { it.multiplier >= bigTh }

                    // Sub-mode 1: Post-Big Wait Interval (e.g., wait 2, 5, 10, 15, 30, 45, 55 min after Big multiplier >= threshold)
                    if (strategy.postBigWaitMinutes > 0 && bigs.isNotEmpty()) {
                        val lastBig = bigs.last()
                        val elapsedMs = (nowMs - lastBig.crashTimestamp).coerceAtLeast(0L)
                        val waitTargetMs = strategy.postBigWaitMinutes * 60_000L
                        val minOcc = strategy.requiredOccurrencesPerHour.coerceAtLeast(1)
                        val periodCutoff = nowMs - (strategy.analysisWindowMinutes.coerceAtLeast(60) * 60_000L)
                        val bigsInPeriod = bigs.count { it.crashTimestamp >= periodCutoff }

                        if (elapsedMs >= waitTargetMs && bigsInPeriod >= minOcc) {
                            val occKey = "timeWait_${strategy.id}_${lastBig.gameId}_${lastBig.dedupKey}_${strategy.postBigWaitMinutes}m"
                            if (strategy.lastTriggeredOccurrenceKey != occKey) {
                                val hourStart = com.example.data.model.CalendarHourUtils.hourStartFor(nowMs)
                                val elapsedMin = elapsedMs / 60_000.0
                                val summary = "${strategy.postBigWaitMinutes}m wait elapsed after ${formatMultiplier(lastBig.multiplier)} (${String.format(Locale.US, "%.1f", elapsedMin)}m ago; $bigsInPeriod in window)"
                                occurrences.add(
                                    com.example.data.model.PatternOccurrenceRecord(
                                        occurrenceKey = occKey,
                                        strategyId = strategy.id,
                                        strategyName = strategy.name,
                                        gameId = lastBig.gameId,
                                        gameName = lastBig.gameName,
                                        patternType = "TIME_INTERVAL_WAIT",
                                        triggerTimestampMs = nowMs,
                                        hourStartTimestamp = hourStart,
                                        detailsSummary = summary,
                                        firstMatchedTimestampMs = lastBig.crashTimestamp,
                                        intervalSincePrevMs = elapsedMs
                                    )
                                )
                                alertEvents.add(
                                    PatternAlertEvent(
                                        strategy = nextStrat,
                                        notificationType = StrategyNotificationType.STRATEGY_REACHED,
                                        title = "${lastBig.gameName}: ${strategy.name}",
                                        message = "TIME INTERVAL: $summary",
                                        triggeringRecordKey = lastBig.dedupKey,
                                        triggeringMultiplier = lastBig.multiplier,
                                        occurrenceKey = occKey,
                                        timestampMs = nowMs
                                    )
                                )
                                nextStrat = nextStrat.copy(lastTriggeredOccurrenceKey = occKey, lastNotifiedAtMs = nowMs, currentlyReached = true)
                            }
                            updatedStrategies.add(nextStrat)
                            continue
                        }
                    }

                    // Sub-mode 2: Consecutive Big gaps within min..max minutes
                    val cond = strategy.getConditions().firstOrNull()
                    val th = cond?.threshold ?: bigTh
                    val gapBigs = gameScoped.filter { it.multiplier >= th }
                    if (gapBigs.size >= 2 && latest.multiplier >= th) {
                        val prevBig = gapBigs[gapBigs.size - 2]
                        val currBig = gapBigs.last()
                        val gapMin = ((currBig.crashTimestamp - prevBig.crashTimestamp).coerceAtLeast(0L)) / 60_000.0
                        if (gapMin in strategy.minIntervalMinutes.toDouble()..strategy.maxIntervalMinutes.toDouble()) {
                            val occKey = "stratB_${strategy.id}_${latest.gameId}_${currBig.dedupKey}"
                            if (strategy.lastTriggeredOccurrenceKey != occKey) {
                                val hourStart = com.example.data.model.CalendarHourUtils.hourStartFor(currBig.crashTimestamp)
                                occurrences.add(
                                    com.example.data.model.PatternOccurrenceRecord(
                                        occurrenceKey = occKey,
                                        strategyId = strategy.id,
                                        strategyName = strategy.name,
                                        gameId = currBig.gameId,
                                        gameName = currBig.gameName,
                                        patternType = strategy.strategyMode.name,
                                        triggerTimestampMs = currBig.crashTimestamp,
                                        hourStartTimestamp = hourStart,
                                        detailsSummary = "${formatMultiplier(currBig.multiplier)} occurred ${String.format(Locale.US, "%.1f", gapMin)}m after ${formatMultiplier(prevBig.multiplier)}",
                                        firstMatchedTimestampMs = prevBig.crashTimestamp,
                                        intervalSincePrevMs = currBig.crashTimestamp - prevBig.crashTimestamp
                                    )
                                )
                                alertEvents.add(
                                    PatternAlertEvent(
                                        strategy = nextStrat,
                                        notificationType = StrategyNotificationType.STRATEGY_REACHED,
                                        title = "${currBig.gameName}: ${strategy.name}",
                                        message = "Big multiplier ${formatMultiplier(currBig.multiplier)} occurred ${String.format(Locale.US, "%.1f", gapMin)}m after ${formatMultiplier(prevBig.multiplier)} (within ${strategy.minIntervalMinutes}–${strategy.maxIntervalMinutes}m).",
                                        triggeringRecordKey = currBig.dedupKey,
                                        triggeringMultiplier = currBig.multiplier,
                                        occurrenceKey = occKey,
                                        timestampMs = nowMs
                                    )
                                )
                                nextStrat = nextStrat.copy(lastTriggeredOccurrenceKey = occKey, lastNotifiedAtMs = nowMs, currentlyReached = true)
                            }
                        } else {
                            nextStrat = nextStrat.copy(currentlyReached = false, currentlyApproaching = false)
                        }
                    } else {
                        nextStrat = nextStrat.copy(currentlyReached = false, currentlyApproaching = false)
                    }
                    updatedStrategies.add(nextStrat)
                }

                com.example.data.model.StrategyMode.STRATEGY_C_COLOR_PATTERNS -> {
                    val targetColors = strategy.getColorSequence()
                    val hexFilters = strategy.getCustomHexSequence()
                    var nextStrat = strategy.copy(lastEvaluatedRecordKey = latest.dedupKey)
                    val dir = strategy.colorMatchDirection

                    val horizontalMatched: List<CrashRecord>? =
                        if ((dir == com.example.data.model.ColorMatchDirection.HORIZONTAL ||
                                dir == com.example.data.model.ColorMatchDirection.BOTH) &&
                            targetColors.isNotEmpty() && gameScoped.size >= targetColors.size
                        ) {
                            val slice = gameScoped.takeLast(targetColors.size)
                            if (matchesColorAndHexSequence(slice, targetColors, hexFilters)) slice else null
                        } else null

                    val cols = strategy.gridColumnsForVertical.coerceIn(2, 12)
                    val verticalMatched: List<CrashRecord>? =
                        if ((dir == com.example.data.model.ColorMatchDirection.VERTICAL ||
                                dir == com.example.data.model.ColorMatchDirection.BOTH) &&
                            targetColors.isNotEmpty() &&
                            gameScoped.size >= ((targetColors.size - 1) * cols + 1)
                        ) {
                            // Top-to-bottom column alignment in a `cols`-column crash grid
                            val verticalSlice = (0 until targetColors.size).map { idx ->
                                val offsetFromNewest = (targetColors.size - 1 - idx) * cols
                                gameScoped[gameScoped.size - 1 - offsetFromNewest]
                            }
                            if (matchesColorAndHexSequence(verticalSlice, targetColors, hexFilters)) verticalSlice else null
                        } else null

                    val matchedSlice = horizontalMatched ?: verticalMatched
                    val matchModeLabel = when {
                        horizontalMatched != null && verticalMatched != null -> "Horizontal & Vertical"
                        verticalMatched != null -> "Vertical (${cols} cols)"
                        else -> "Horizontal"
                    }

                    if (matchedSlice != null) {
                        val anchor = matchedSlice.first()
                        val occKey = "stratC_${strategy.id}_${latest.gameId}_${anchor.dedupKey}_${latest.dedupKey}"
                        if (strategy.lastTriggeredOccurrenceKey != occKey) {
                            val hourStart = com.example.data.model.CalendarHourUtils.hourStartFor(latest.crashTimestamp)
                            val seqDesc = targetColors.joinToString(" -> ") { it.displayName }
                            val hexNote = matchedSlice.joinToString(", ") { it.detectedColorHex }
                            occurrences.add(
                                com.example.data.model.PatternOccurrenceRecord(
                                    occurrenceKey = occKey,
                                    strategyId = strategy.id,
                                    strategyName = strategy.name,
                                    gameId = latest.gameId,
                                    gameName = latest.gameName,
                                    patternType = "${strategy.strategyMode.name}_$matchModeLabel",
                                    triggerTimestampMs = latest.crashTimestamp,
                                    hourStartTimestamp = hourStart,
                                    detailsSummary = "Observed $matchModeLabel color sequence: $seqDesc ($hexNote)",
                                    firstMatchedTimestampMs = anchor.crashTimestamp
                                )
                            )
                            alertEvents.add(
                                PatternAlertEvent(
                                    strategy = nextStrat,
                                    notificationType = StrategyNotificationType.STRATEGY_REACHED,
                                    title = "${latest.gameName}: ${strategy.name}",
                                    message = "Observed $matchModeLabel color sequence $seqDesc (latest ${formatMultiplier(latest.multiplier)} · ${latest.detectedColorHex}).",
                                    triggeringRecordKey = latest.dedupKey,
                                    triggeringMultiplier = latest.multiplier,
                                    occurrenceKey = occKey,
                                    timestampMs = nowMs
                                )
                            )
                            nextStrat = nextStrat.copy(lastTriggeredOccurrenceKey = occKey, lastNotifiedAtMs = nowMs, currentlyReached = true)
                        }
                    } else {
                        nextStrat = nextStrat.copy(currentlyReached = false, currentlyApproaching = false)
                    }
                    updatedStrategies.add(nextStrat)
                }

                com.example.data.model.StrategyMode.CUSTOM_CONDITION -> {
                    run {
                        // Convert CrashRecords to MultiplierRecords so general custom threshold/sequence/between conditions evaluate directly
                        val multEquiv = gameScoped.map { cr ->
                            MultiplierRecord(
                                id = cr.id,
                                gameId = cr.gameId,
                                sessionId = cr.sessionId,
                                roundId = cr.roundIdentifier,
                                multiplier = cr.multiplier,
                                observationTimestampMs = cr.crashTimestamp,
                                crashTimestamp = cr.crashTimestamp,
                                platform = cr.gameName,
                                dedupKey = cr.dedupKey
                            )
                        }
                        val subBatch = evaluateAllStrategies(listOf(strategy), multEquiv, nowMs = nowMs)
                        val upd = subBatch.updatedStrategies.firstOrNull() ?: strategy
                        subBatch.events.filter { it.notificationType == StrategyNotificationType.STRATEGY_REACHED }.forEach { ev ->
                            val hourStart = com.example.data.model.CalendarHourUtils.hourStartFor(ev.timestampMs)
                            occurrences.add(
                                com.example.data.model.PatternOccurrenceRecord(
                                    occurrenceKey = ev.occurrenceKey,
                                    strategyId = strategy.id,
                                    strategyName = strategy.name,
                                    gameId = latest.gameId,
                                    gameName = latest.gameName,
                                    patternType = strategy.strategyMode.name,
                                    triggerTimestampMs = latest.crashTimestamp,
                                    hourStartTimestamp = hourStart,
                                    detailsSummary = ev.message,
                                    firstMatchedTimestampMs = latest.crashTimestamp
                                )
                            )
                        }
                        alertEvents.addAll(subBatch.events)
                        updatedStrategies.add(upd)
                    }
                }
            }
        }

        return CrashPatternEvaluationBatch(
            updatedStrategies = updatedStrategies,
            events = alertEvents,
            newOccurrences = occurrences
        )
    }

    /**
     * Evaluates time-based strategies even between crash rounds (post-Big elapsed countdowns and HH:mm specific alert times).
     */
    fun evaluateTimeSchedulesTick(
        strategies: List<CustomPatternStrategy>,
        crashRecords: List<CrashRecord>,
        bigThreshold: Double = 10.00,
        nowMs: Long = System.currentTimeMillis()
    ): CrashPatternEvaluationBatch {
        val timeStrategies = strategies.filter {
            it.isEnabled && (
                it.strategyGroup == com.example.data.model.StrategyGroup.TIME_INTERVAL ||
                    it.strategyMode == com.example.data.model.StrategyMode.STRATEGY_B_BIG_INTERVALS
                )
        }
        if (timeStrategies.isEmpty()) {
            return CrashPatternEvaluationBatch(strategies, emptyList(), emptyList())
        }

        val cal = java.util.Calendar.getInstance().apply { timeInMillis = nowMs }
        val currentHm = String.format(
            Locale.US,
            "%02d:%02d",
            cal.get(java.util.Calendar.HOUR_OF_DAY),
            cal.get(java.util.Calendar.MINUTE)
        )
        val dayKey = String.format(
            Locale.US,
            "%04d%02d%02d_%02d%02d",
            cal.get(java.util.Calendar.YEAR),
            cal.get(java.util.Calendar.MONTH) + 1,
            cal.get(java.util.Calendar.DAY_OF_MONTH),
            cal.get(java.util.Calendar.HOUR_OF_DAY),
            cal.get(java.util.Calendar.MINUTE)
        )

        val updatedMap = strategies.associateBy { it.id }.toMutableMap()
        val events = mutableListOf<PatternAlertEvent>()
        val occs = mutableListOf<com.example.data.model.PatternOccurrenceRecord>()

        for (strategy in timeStrategies) {
            var nextStrat = strategy
            val scoped = if (strategy.targetGameId == "ALL") crashRecords else crashRecords.filter { it.gameId == strategy.targetGameId }
            val bigTh = if (strategy.bigMultiplierThreshold > 1.0) strategy.bigMultiplierThreshold else bigThreshold
            val bigs = scoped.filter { it.multiplier >= bigTh }.sortedBy { it.crashTimestamp }

            // Check Post-Big wait timer
            if (strategy.postBigWaitMinutes > 0 && bigs.isNotEmpty()) {
                val lastBig = bigs.last()
                val elapsed = (nowMs - lastBig.crashTimestamp).coerceAtLeast(0L)
                if (elapsed >= strategy.postBigWaitMinutes * 60_000L) {
                    val occKey = "timeWait_${strategy.id}_${lastBig.gameId}_${lastBig.dedupKey}_${strategy.postBigWaitMinutes}m"
                    if (strategy.lastTriggeredOccurrenceKey != occKey) {
                        val hourStart = com.example.data.model.CalendarHourUtils.hourStartFor(nowMs)
                        val summary = "${strategy.postBigWaitMinutes}m wait complete after ${formatMultiplier(lastBig.multiplier)} in ${lastBig.gameName}"
                        occs.add(
                            com.example.data.model.PatternOccurrenceRecord(
                                occurrenceKey = occKey,
                                strategyId = strategy.id,
                                strategyName = strategy.name,
                                gameId = lastBig.gameId,
                                gameName = lastBig.gameName,
                                patternType = "TIME_INTERVAL_WAIT",
                                triggerTimestampMs = nowMs,
                                hourStartTimestamp = hourStart,
                                detailsSummary = summary,
                                firstMatchedTimestampMs = lastBig.crashTimestamp,
                                intervalSincePrevMs = elapsed
                            )
                        )
                        events.add(
                            PatternAlertEvent(
                                strategy = nextStrat,
                                notificationType = StrategyNotificationType.STRATEGY_REACHED,
                                title = "Time Alert: ${strategy.name}",
                                message = summary,
                                triggeringRecordKey = lastBig.dedupKey,
                                triggeringMultiplier = lastBig.multiplier,
                                occurrenceKey = occKey,
                                timestampMs = nowMs
                            )
                        )
                        nextStrat = nextStrat.copy(lastTriggeredOccurrenceKey = occKey, lastNotifiedAtMs = nowMs, currentlyReached = true)
                    }
                }
            }

            // Check specific HH:mm scheduled alerts
            val scheduledTimes = strategy.getSpecificAlertTimes()
                .map { (h, m) -> String.format(Locale.US, "%02d:%02d", h, m) }
            if (scheduledTimes.contains(currentHm)) {
                val occKey = "specTime_${strategy.id}_$dayKey"
                if (strategy.lastTriggeredOccurrenceKey != occKey) {
                    val hourStart = com.example.data.model.CalendarHourUtils.hourStartFor(nowMs)
                    val gId = strategy.targetGameId.takeIf { it != "ALL" } ?: scoped.lastOrNull()?.gameId ?: "unknown"
                    val gName = scoped.lastOrNull()?.gameName ?: "All Games"
                    occs.add(
                        com.example.data.model.PatternOccurrenceRecord(
                            occurrenceKey = occKey,
                            strategyId = strategy.id,
                            strategyName = strategy.name,
                            gameId = gId,
                            gameName = gName,
                            patternType = "TIME_SCHEDULE_HM",
                            triggerTimestampMs = nowMs,
                            hourStartTimestamp = hourStart,
                            detailsSummary = "Scheduled time alert at $currentHm",
                            firstMatchedTimestampMs = nowMs
                        )
                    )
                    events.add(
                        PatternAlertEvent(
                            strategy = nextStrat,
                            notificationType = StrategyNotificationType.STRATEGY_REACHED,
                            title = "Scheduled Time Alert: ${strategy.name}",
                            message = "Configured alert time $currentHm reached for ${strategy.name}.",
                            occurrenceKey = occKey,
                            timestampMs = nowMs
                        )
                    )
                    nextStrat = nextStrat.copy(lastTriggeredOccurrenceKey = occKey, lastNotifiedAtMs = nowMs)
                }
            }

            updatedMap[strategy.id] = nextStrat
        }

        return CrashPatternEvaluationBatch(
            updatedStrategies = strategies.map { updatedMap[it.id] ?: it },
            events = events,
            newOccurrences = occs
        )
    }

    private fun matchesColorAndHexSequence(
        slice: List<CrashRecord>,
        targetColors: List<com.example.data.model.CrashColorCategory>,
        hexFilters: List<String>
    ): Boolean {
        if (slice.size != targetColors.size) return false
        return slice.indices.all { idx ->
            val rec = slice[idx]
            val catOk = rec.colorCategory == targetColors[idx]
            val expectedHex = hexFilters.getOrNull(idx)?.trim().orEmpty()
            val hexOk = expectedHex.isBlank() || expectedHex.equals("ANY", ignoreCase = true) ||
                rec.detectedColorHex.equals(expectedHex, ignoreCase = true)
            catOk && hexOk
        }
    }
}

data class CrashPatternEvaluationBatch(
    val updatedStrategies: List<CustomPatternStrategy>,
    val events: List<PatternAlertEvent>,
    val newOccurrences: List<com.example.data.model.PatternOccurrenceRecord>
)

enum class PatternComparator(val symbol: String, val label: String) {
    LESS_THAN("<", "Below (< exclusive)"),
    LESS_OR_EQUAL("<=", "At or Below (<= inclusive)"),
    GREATER_THAN(">", "Above (> exclusive)"),
    GREATER_OR_EQUAL(">=", "At or Above (>= inclusive)"),
    BETWEEN("between", "Between Range");

    fun matches(value: Double, threshold: Double, upperThreshold: Double? = null): Boolean {
        if (!value.isFinite() || !threshold.isFinite()) return false
        return when (this) {
            LESS_THAN -> value < threshold
            LESS_OR_EQUAL -> value <= threshold
            GREATER_THAN -> value > threshold
            GREATER_OR_EQUAL -> value >= threshold
            BETWEEN -> {
                val upper = upperThreshold?.takeIf { it.isFinite() && it >= threshold } ?: threshold
                value in threshold..upper
            }
        }
    }
}

    AND("ALL conditions match (AND)"),
    OR("ANY condition matches (OR)")
}

data class PatternCondition(
    val comparator: PatternComparator = PatternComparator.LESS_THAN,
    val threshold: Double = 2.00,
    val upperThreshold: Double? = null,
    val consecutiveRounds: Int = 5,
    val timeWindowMinutes: Int? = null,
    val sequenceMultipliers: List<Double> = emptyList()
) {
    fun summaryText(): String {
        val multFormatted = String.format(Locale.US, "%.2f", threshold)
        val rangeText = if (comparator == PatternComparator.BETWEEN && upperThreshold != null) {
            val upperFormatted = String.format(Locale.US, "%.2f", upperThreshold)
            "${multFormatted}x–${upperFormatted}x"
        } else {
            "${comparator.symbol} ${multFormatted}x"
        }
        val windowSuffix = if (timeWindowMinutes != null && timeWindowMinutes > 0) {
            " within ${timeWindowMinutes}m"
        } else {
            ""
        }
        return if (sequenceMultipliers.isNotEmpty()) {
            "Sequence [${sequenceMultipliers.joinToString(", ") { "${String.format(Locale.US, "%.2f", it)}x" }}]$windowSuffix"
        } else {
            "$consecutiveRounds consecutive rounds $rangeText$windowSuffix"
        }
    }
}

@Entity(tableName = "custom_pattern_strategies")
data class CustomPatternStrategy(
    @PrimaryKey(autoGenerate = true)
    val id: Long = 0,
    val name: String,
    val isEnabled: Boolean = true,
    val strategyGroup: StrategyGroup = StrategyGroup.MULTIPLIER_INTERVAL,
    val strategyMode: StrategyMode = StrategyMode.CUSTOM_CONDITION,
    val multiplierRuleSubtype: MultiplierRuleSubtype = MultiplierRuleSubtype.STANDARD_THRESHOLD,
    val logicOperator: PatternLogicOperator = PatternLogicOperator.AND,
    val conditionsJson: String = serializeConditions(emptyList()),
    // Composite ABOVE X (N) -> BELOW Y (M) fields
    val aboveThresholdX: Double = 2.00,
    val aboveConsecutiveCount: Int = 3,
    val belowThresholdY: Double = 2.00,
    val belowConsecutiveCount: Int = 2,
    val inclusiveBounds: Boolean = false,
    // Time Interval fields (post-big wait + 1h/3h/6h/custom frequency + specific alert times)
    val postBigWaitMinutes: Int = 5,
    val bigMultiplierThreshold: Double = 10.00,
    val specificAlertTimesCsv: String = "",
    // Color Sequence fields (horizontal / vertical + custom HEX palette)
    val colorMatchDirection: ColorMatchDirection = ColorMatchDirection.HORIZONTAL,
    val gridColumnsForVertical: Int = 8,
    val customColorHexFilterCsv: String = "",
    // Common alerts, cooldown, ordering & repeats
    val waitingPeriodSeconds: Int = 0,
    val notificationLevel: NotificationPriorityLevel = NotificationPriorityLevel.HIGH,
    val approachingEnabled: Boolean = true,
    val customMessage: String = "",
    val repeatPolicy: RepeatNotificationPolicy = RepeatNotificationPolicy.ONCE_PER_OCCURRENCE,
    val repeatCooldownMinutes: Int = 0,
    val repeatLimit: Int = 0,
    val notificationEnabled: Boolean = true,
    val useAlarmStyle: Boolean = true,
    val vibrationEnabled: Boolean = true,
    val sortOrder: Int = 0,
    val requiredOccurrencesPerHour: Int = 2,
    val analysisWindowMinutes: Int = 60,
    val minIntervalMinutes: Int = 5,
    val maxIntervalMinutes: Int = 15,
    val colorSequenceCsv: String = "",
    val targetGameId: String = "ALL",
    val lastEvaluatedRecordKey: String? = null,
    val lastTriggeredOccurrenceKey: String? = null,
    val lastApproachedOccurrenceKey: String? = null,
    val lastNotifiedAtMs: Long = 0L,
    val currentlyApproaching: Boolean = false,
    val currentlyReached: Boolean = false,
    val createdAtMs: Long = System.currentTimeMillis()
) {
    fun effectiveGroup(): StrategyGroup = when (strategyMode) {
        StrategyMode.STRATEGY_B_BIG_INTERVALS -> StrategyGroup.TIME_INTERVAL
        StrategyMode.STRATEGY_C_COLOR_PATTERNS -> StrategyGroup.COLOR_SEQUENCE
        else -> strategyGroup
    }

    fun resolveAlarmProfileCategory(): AlarmProfileCategory {
        if (!useAlarmStyle) return AlarmProfileCategory.STANDARD_ALERTS
        return when (effectiveGroup()) {
            StrategyGroup.TIME_INTERVAL -> AlarmProfileCategory.TIME_BASED_ALERTS
            StrategyGroup.COLOR_SEQUENCE -> AlarmProfileCategory.COLOR_SEQUENCE_ALERTS
            StrategyGroup.MULTIPLIER_INTERVAL -> {
                if (multiplierRuleSubtype != MultiplierRuleSubtype.STANDARD_THRESHOLD) {
                    AlarmProfileCategory.MULTIPLIER_CONDITIONS
                } else if (notificationLevel == NotificationPriorityLevel.URGENT) {
                    AlarmProfileCategory.HIGH_PRIORITY_ALERTS
                } else {
                    AlarmProfileCategory.MULTIPLIER_CONDITIONS
                }
            }
        }
    }

    fun getConditions(): List<PatternCondition> = deserializeConditions(conditionsJson)

    fun withConditions(conditions: List<PatternCondition>): CustomPatternStrategy {
        return copy(conditionsJson = serializeConditions(conditions))
    }

    fun getColorSequence(): List<CrashColorCategory> {
        if (colorSequenceCsv.isBlank()) return emptyList()
        return colorSequenceCsv.split(",")
            .mapNotNull { token ->
                val clean = token.trim()
                CrashColorCategory.entries.firstOrNull {
                    it.name.equals(clean, ignoreCase = true) ||
                        it.displayName.equals(clean, ignoreCase = true)
                }
            }
    }

    fun getCustomHexSequence(): List<String> {
        if (customColorHexFilterCsv.isBlank()) return emptyList()
        return customColorHexFilterCsv.split(",")
            .map { it.trim() }
            .filter { it.startsWith("#") && (it.length == 7 || it.length == 9) }
            .map { ColorClassifier.normalizeHex(it) }
    }

    fun getSpecificAlertTimes(): List<Pair<Int, Int>> {
        if (specificAlertTimesCsv.isBlank()) return emptyList()
        return specificAlertTimesCsv.split(",")
            .mapNotNull { token ->
                val parts = token.trim().split(":")
                if (parts.size != 2) return@mapNotNull null
                val h = parts[0].toIntOrNull()?.takeIf { it in 0..23 } ?: return@mapNotNull null
                val m = parts[1].toIntOrNull()?.takeIf { it in 0..59 } ?: return@mapNotNull null
                h to m
            }
    }

    fun summaryDescription(): String {
        if (multiplierRuleSubtype == MultiplierRuleSubtype.ABOVE_THEN_BELOW) {
            val aboveOp = if (inclusiveBounds) ">=" else ">"
            val belowOp = if (inclusiveBounds) "<=" else "<"
            return "After ${aboveConsecutiveCount}x $aboveOp ${String.format(Locale.US, "%.2f", aboveThresholdX)}x, " +
                "then ${belowConsecutiveCount}x $belowOp ${String.format(Locale.US, "%.2f", belowThresholdY)}x -> Ring Alarm"
        }
        return when (strategyMode) {
            StrategyMode.STRATEGY_A_CONSECUTIVE_LOWS -> {
                val c = getConditions().firstOrNull()
                val th = c?.threshold ?: 2.00
                val count = c?.consecutiveRounds ?: 14
                val op = c?.comparator?.symbol ?: if (inclusiveBounds) "<=" else "<"
                "$count consecutive $op ${String.format(Locale.US, "%.2f", th)}x · ${requiredOccurrencesPerHour}/${analysisWindowMinutes}m"
            }
            StrategyMode.STRATEGY_B_BIG_INTERVALS -> {
                val th = bigMultiplierThreshold.takeIf { it >= 1.0 }
                    ?: getConditions().firstOrNull()?.threshold
                    ?: 10.00
                val waitPart = if (postBigWaitMinutes > 0) "Wait ${postBigWaitMinutes}m after >=${String.format(Locale.US, "%.2f", th)}x" else "Gap ${minIntervalMinutes}–${maxIntervalMinutes}m"
                val schedPart = if (specificAlertTimesCsv.isNotBlank()) " @ $specificAlertTimesCsv" else ""
                "$waitPart · ${requiredOccurrencesPerHour}x/${analysisWindowMinutes}m$schedPart"
            }
            StrategyMode.STRATEGY_C_COLOR_PATTERNS -> {
                val colors = getColorSequence().joinToString(" -> ") { it.displayName }
                    .ifBlank { customColorHexFilterCsv.ifBlank { colorSequenceCsv } }
                "${colorMatchDirection.name}: $colors (${analysisWindowMinutes}m)"
            }
            StrategyMode.CUSTOM_CONDITION -> {
                val conds = getConditions()
                if (conds.isEmpty()) return "No conditions configured"
                val joined = conds.joinToString(if (logicOperator == PatternLogicOperator.AND) " AND " else " OR ") {
                    it.summaryText()
                }
                if (waitingPeriodSeconds > 0) "$joined · Wait ${waitingPeriodSeconds}s" else joined
            }
        }
    }

    companion object {
        fun serializeConditions(conditions: List<PatternCondition>): String {
            if (conditions.isEmpty()) return ""
            return conditions.joinToString(";") { c ->
                val seqStr = c.sequenceMultipliers.joinToString(",") {
                    String.format(Locale.US, "%.2f", it)
                }
                listOf(
                    c.comparator.name,
                    c.threshold.toString(),
                    c.upperThreshold?.toString() ?: "",
                    c.consecutiveRounds.coerceAtLeast(1).toString(),
                    c.timeWindowMinutes?.toString() ?: "",
                    seqStr
                ).joinToString("|")
            }
        }

        fun deserializeConditions(encoded: String?): List<PatternCondition> {
            if (encoded.isNullOrBlank()) return emptyList()
            return encoded.split(";").mapNotNull { entry ->
                val parts = entry.split("|")
                if (parts.size < 4) return@mapNotNull null
                val comp = runCatching { PatternComparator.valueOf(parts[0]) }.getOrNull()
                    ?: return@mapNotNull null
                val threshold = parts[1].toDoubleOrNull()?.takeIf { it.isFinite() && it >= 1.0 }
                    ?: return@mapNotNull null
                val upper = parts.getOrNull(2)?.toDoubleOrNull()?.takeIf { it.isFinite() && it >= threshold }
                val consecutive = parts.getOrNull(3)?.toIntOrNull()?.coerceAtLeast(1) ?: 1
                val timeWindow = parts.getOrNull(4)?.toIntOrNull()?.takeIf { it > 0 }
                val seq = parts.getOrNull(5)
                    ?.split(",")
                    ?.mapNotNull { it.trim().toDoubleOrNull()?.takeIf { v -> v.isFinite() && v >= 1.0 } }
                    ?: emptyList()
                PatternCondition(
                    comparator = comp,
                    threshold = threshold,
                    upperThreshold = upper,
                    consecutiveRounds = consecutive,
                    timeWindowMinutes = timeWindow,
                    sequenceMultipliers = seq
                )
            }
        }
    }
}

data class StrategyValidationResult(
    val isValid: Boolean,
    val warnings: List<String> = emptyList(),
    val blockingErrors: List<String> = emptyList()
)

object StrategyRangeValidator {
    fun validate(
        candidate: CustomPatternStrategy,
        existing: List<CustomPatternStrategy>,
        configuredLowThreshold: Double = 2.00,
        configuredBigThreshold: Double = 10.00
    ): StrategyValidationResult {
        val errors = mutableListOf<String>()
        val warnings = mutableListOf<String>()

        if (candidate.name.isBlank()) {
            errors.add("Strategy name cannot be empty.")
        }
        if (configuredLowThreshold >= configuredBigThreshold) {
            warnings.add("Global LOW threshold (${configuredLowThreshold}x) >= BIG threshold (${configuredBigThreshold}x) causes range overlap.")
        }

        val conds = candidate.getConditions()
        if (candidate.multiplierRuleSubtype == MultiplierRuleSubtype.ABOVE_THEN_BELOW) {
            if (candidate.aboveThresholdX < 1.0 || candidate.belowThresholdY < 1.0) {
                errors.add("Both Above X and Below Y thresholds must be >= 1.00x.")
            }
            if (candidate.aboveConsecutiveCount < 1 || candidate.belowConsecutiveCount < 1) {
                errors.add("Consecutive counts must be at least 1.")
            }
            if (candidate.belowThresholdY > candidate.aboveThresholdX) {
                warnings.add("Below threshold (${candidate.belowThresholdY}x) is higher than Above threshold (${candidate.aboveThresholdX}x) — overlapping zone will trigger on mid-range values.")
            }
        } else if (candidate.effectiveGroup() == StrategyGroup.MULTIPLIER_INTERVAL) {
            if (conds.isEmpty()) {
                errors.add("At least one multiplier condition is required.")
            }
            conds.forEachIndexed { idx, cond ->
                if (cond.threshold < 1.0) {
                    errors.add("Condition #${idx + 1} threshold must be >= 1.00x.")
                }
                if (cond.comparator == PatternComparator.BETWEEN) {
                    val up = cond.upperThreshold
                    if (up == null || up <= cond.threshold) {
                        errors.add("Condition #${idx + 1} Between upper bound must exceed lower bound.")
                    }
                }
            }
            for (i in conds.indices) {
                for (j in i + 1 until conds.size) {
                    val a = conds[i]
                    val b = conds[j]
                    val overlaps = when {
                        a.comparator in listOf(PatternComparator.LESS_THAN, PatternComparator.LESS_OR_EQUAL) &&
                            b.comparator in listOf(PatternComparator.GREATER_THAN, PatternComparator.GREATER_OR_EQUAL) &&
                            a.threshold > b.threshold -> true
                        b.comparator in listOf(PatternComparator.LESS_THAN, PatternComparator.LESS_OR_EQUAL) &&
                            a.comparator in listOf(PatternComparator.GREATER_THAN, PatternComparator.GREATER_OR_EQUAL) &&
                            b.threshold > a.threshold -> true
                        else -> false
                    }
                    if (overlaps) {
                        warnings.add("Conditions #${i + 1} and #${j + 1} have overlapping multiplier ranges.")
                    }
                }
            }
        } else if (candidate.effectiveGroup() == StrategyGroup.TIME_INTERVAL) {
            if (candidate.bigMultiplierThreshold < 1.5) {
                warnings.add("BIG threshold (${candidate.bigMultiplierThreshold}x) is unusually low; consider >= 5.00x.")
            }
            if (candidate.analysisWindowMinutes < 5) {
                errors.add("Analysis time interval must be at least 5 minutes.")
            }
            if (candidate.minIntervalMinutes > candidate.maxIntervalMinutes && candidate.maxIntervalMinutes > 0) {
                errors.add("Minimum interval minutes cannot exceed maximum interval minutes.")
            }
        } else if (candidate.effectiveGroup() == StrategyGroup.COLOR_SEQUENCE) {
            if (candidate.getColorSequence().isEmpty() && candidate.getCustomHexSequence().isEmpty()) {
                errors.add("Select at least one color or custom HEX color in the sequence.")
            }
        }

        // Check conflict with other active strategies targeting the same game
        existing.filter { it.id != candidate.id && it.isEnabled }.forEach { other ->
            val sameGame = other.targetGameId == candidate.targetGameId ||
                other.targetGameId == "ALL" || candidate.targetGameId == "ALL"
            if (sameGame && other.effectiveGroup() == candidate.effectiveGroup() &&
                other.multiplierRuleSubtype == candidate.multiplierRuleSubtype &&
                other.summaryDescription() == candidate.summaryDescription()
            ) {
                warnings.add("Duplicates active rule '${other.name}' for target ${other.targetGameId}.")
            }
        }

        return StrategyValidationResult(
            isValid = errors.isEmpty(),
            warnings = warnings.distinct(),
            blockingErrors = errors.distinct()
        )
    }
}

/**
 * Properly managed audio & vibration controller for active signal alarms and settings tests.
 * Respects system Do Not Disturb, audio focus / volume, and configured max duration;
 * stops and releases MediaPlayer resources immediately upon Stop, Snooze, or timeout.
 */
object AlarmAudioController {
    private const val TAG = "AlarmAudioController"
    private val scope = CoroutineScope(Dispatchers.Default + SupervisorJob())

    private val _isAlarmRinging = MutableStateFlow(false)
    val isAlarmRinging: StateFlow<Boolean> = _isAlarmRinging.asStateFlow()

    private val _activeAlarmTitle = MutableStateFlow<String?>(null)
    val activeAlarmTitle: StateFlow<String?> = _activeAlarmTitle.asStateFlow()

    private val _snoozedUntilMs = MutableStateFlow(0L)
    val snoozedUntilMs: StateFlow<Long> = _snoozedUntilMs.asStateFlow()

    private val _lastAudioStatus = MutableStateFlow<String?>(null)
    val lastAudioStatus: StateFlow<String?> = _lastAudioStatus.asStateFlow()

    private var mediaPlayer: MediaPlayer? = null
    private var autoStopJob: Job? = null
    private var activeVibrator: Vibrator? = null
    @Volatile
    private var lastAppContext: Context? = null

    fun vibrationTimingsFor(patternKey: String): LongArray = when (patternKey.uppercase()) {
        "SHORT_PULSE" -> longArrayOf(0L, 120L, 100L, 120L)
        "DISTINCTIVE" -> longArrayOf(0L, 350L, 150L, 350L, 150L, 600L)
        "LONG_CONTINUOUS" -> longArrayOf(0L, 800L, 250L, 800L)
        "NONE" -> longArrayOf(0L)
        else -> longArrayOf(0L, 220L, 120L, 220L)
    }

    /**
     * Resolves a user-selected Android sound URI ONLY.
     * Never silently auto-assigns a sound when customUriString is empty.
     */
    fun resolveSoundUri(customUriString: String, isAlarm: Boolean): Uri? {
        if (customUriString.isBlank()) return null
        return runCatching { Uri.parse(customUriString.trim()) }.getOrNull()
    }

    fun previewSoundUri(context: Context, uriString: String, asAlarmStream: Boolean = true) {
        val uri = resolveSoundUri(uriString, isAlarm = asAlarmStream)
        if (uri == null) {
            _lastAudioStatus.value = "No sound selected — tap Choose Sound to pick one."
            return
        }
        stopAlarm()
        runCatching {
            val usage = if (asAlarmStream) AudioAttributes.USAGE_ALARM else AudioAttributes.USAGE_NOTIFICATION_EVENT
            val player = MediaPlayer().apply {
                setAudioAttributes(
                    AudioAttributes.Builder()
                        .setUsage(usage)
                        .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
                        .build()
                )
                setDataSource(context.applicationContext, uri)
                isLooping = false
                setOnCompletionListener { mp ->
                    runCatching { mp.release() }
                    if (mediaPlayer === mp) mediaPlayer = null
                }
                prepare()
                start()
            }
            mediaPlayer = player
            _lastAudioStatus.value = "Previewing selected sound."
        }.onFailure {
            _lastAudioStatus.value = "Unable to play selected sound URI."
        }
    }

    fun testAlarmProfile(context: Context, profile: AlarmProfileEntity) {
        val settings = NotificationSettings(
            highPriorityAlarmEnabled = true,
            alarmSoundUri = profile.soundUri,
            alarmVibrationPattern = if (profile.vibrationEnabled) profile.vibrationPattern else "NONE",
            alarmRepeated = profile.repeatEnabled,
            alarmMaxDurationMinutes = 1
        )
        clearSnooze()
        if (profile.soundUri.isBlank() && !profile.vibrationEnabled) {
            _lastAudioStatus.value = "Profile '${profile.category().displayTitle}' has no sound or vibration selected."

    }

    fun isAlarmSnoozed(nowMs: Long = System.currentTimeMillis()): Boolean =
        _snoozedUntilMs.value > nowMs

    const val ALARM_NOTIFICATION_ID = 9999
    const val ACTION_STOP_ALARM = "com.example.action.STOP_ALARM"

    fun startAlarm(
        context: Context?,
        settings: NotificationSettings,
        title: String = "Signal Detected — Pattern Match",
        nowMs: Long = System.currentTimeMillis()
    ) {
        if (!settings.highPriorityAlarmEnabled) {
            _lastAudioStatus.value = "Signal alarm disabled in profile."
            return
        }
        if (isAlarmSnoozed(nowMs)) {
            _lastAudioStatus.value = "Signal alarm snoozed."
            return
        }
        if (context == null) {
            _isAlarmRinging.value = true
            _activeAlarmTitle.value = title
            return
        }
        if (AndroidSystemNotificationDispatcher.isDoNotDisturbBlocking(context, NotificationPriorityLevel.URGENT)) {
            _lastAudioStatus.value = "Do Not Disturb active on device; alarm audio suppressed."
            return
        }

        val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val alarmVol = audioManager?.getStreamVolume(AudioManager.STREAM_ALARM) ?: 1
        if (alarmVol <= 0) {
            _lastAudioStatus.value = "Device alarm volume is muted (0); triggering vibration only."
        }

        lastAppContext = context.applicationContext
        stopAlarm(context)
        _isAlarmRinging.value = true
        _activeAlarmTitle.value = title

        runCatching {
            val nm = context.getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                AndroidSystemNotificationDispatcher.ensureAllChannels(context, settings)
            }
            val stopIntent = Intent(context, AlarmStopReceiver::class.java).apply {
                action = ACTION_STOP_ALARM
            }
            val stopActionPendingIntent = PendingIntent.getBroadcast(
                context,
                2002,
                stopIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
            )
            val stopTapPendingIntent = PendingIntent.getBroadcast(
                context,
                2003,
                stopIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
            )
            val stopDeletePendingIntent = PendingIntent.getBroadcast(
                context,
                2004,
                stopIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
            )

            val channelId = AndroidSystemNotificationDispatcher.resolveVersionedChannelId(
                AndroidSystemNotificationDispatcher.CHANNEL_ALARM, settings
            )
            val builder = NotificationCompat.Builder(context, channelId)
                .setSmallIcon(R.mipmap.ic_launcher_round)
                .setContentTitle(title)
                .setContentText("Alarm ringing — tap STOP ALARM or swipe away to silence.")
                .setPriority(NotificationCompat.PRIORITY_MAX)
                .setAutoCancel(true)
                .setSilent(true)
                .setContentIntent(stopTapPendingIntent)
                .setDeleteIntent(stopDeletePendingIntent)
                .addAction(R.mipmap.ic_launcher_round, "STOP ALARM", stopActionPendingIntent)

            nm?.notify(ALARM_NOTIFICATION_ID, builder.build())
        }

        val uri = resolveSoundUri(settings.alarmSoundUri, isAlarm = true)
        if (uri != null && alarmVol > 0) {
            runCatching {
                val player = MediaPlayer().apply {
                    setAudioAttributes(
                        AudioAttributes.Builder()
                            .setUsage(AudioAttributes.USAGE_ALARM)
                            .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
                            .build()
                    )
                    setDataSource(context.applicationContext, uri)
                    isLooping = settings.alarmRepeated
                    setOnCompletionListener {
                        if (!settings.alarmRepeated) {
                            stopAlarm(context)
                        }
                    }
                    setOnErrorListener { _, _, _ ->
                        stopAlarm(context)
                        true
                    }
                    prepare()
                    start()
                }
                mediaPlayer = player
            }.onFailure {
                Log.w(TAG, "Alarm MediaPlayer failed; keeping UI alarm state", it)
                _lastAudioStatus.value = "Unable to play alarm URI on this device."
            }
        }

        if (settings.alarmVibrationPattern != "NONE") {
            triggerVibration(context, settings.alarmVibrationPattern, repeat = settings.alarmRepeated)
        }

        val maxDurationMs = settings.alarmMaxDurationMinutes.coerceIn(1, 30) * 60_000L
        autoStopJob?.cancel()
        autoStopJob = scope.launch {
            delay(maxDurationMs)
            stopAlarm(context)
        }
    }

    fun stopAlarm(context: Context? = null) {
        autoStopJob?.cancel()
        autoStopJob = null
        runCatching {
            mediaPlayer?.apply {
                if (isPlaying) stop()
                reset()
                release()
            }
        }
        mediaPlayer = null
        runCatching { activeVibrator?.cancel() }
        activeVibrator = null
        _isAlarmRinging.value = false
        _activeAlarmTitle.value = null

        val ctx = context ?: lastAppContext ?: com.example.service.ScreenEarnerAccessibilityService.instance?.applicationContext
        if (ctx != null) {
            runCatching {
                val nm = ctx.getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager
                nm?.cancel(ALARM_NOTIFICATION_ID)
            }
        }
    }

    fun snoozeAlarm(snoozeMinutes: Int = 5, nowMs: Long = System.currentTimeMillis()) {
        stopAlarm()

        _snoozedUntilMs.value = nowMs + mins * 60_000L
        _lastAudioStatus.value = "Signal alarm snoozed for ${mins}m."
    }

    @Suppress("DEPRECATION")
    private fun triggerVibration(context: Context, patternKey: String, repeat: Boolean) {
        if (patternKey == "NONE") return
        val timings = vibrationTimingsFor(patternKey)
        if (timings.size <= 1) return
        val vibrator = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            val vm = context.getSystemService(Context.VIBRATOR_MANAGER_SERVICE) as? VibratorManager
            vm?.defaultVibrator
        } else {
            context.getSystemService(Context.VIBRATOR_SERVICE) as? Vibrator
        } ?: return
        activeVibrator = vibrator
        runCatching {
            val repeatIndex = if (repeat) 0 else -1
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                vibrator.vibrate(VibrationEffect.createWaveform(timings, repeatIndex))
            } else {
                vibrator.vibrate(timings, repeatIndex)
            }
        }
    }

    fun openChannelSettings(context: Context, channelId: String) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val intent = Intent(Settings.ACTION_CHANNEL_NOTIFICATION_SETTINGS).apply {
                putExtra(Settings.EXTRA_APP_PACKAGE, context.packageName)
                putExtra(Settings.EXTRA_CHANNEL_ID, channelId)
                addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            }
            runCatching { context.startActivity(intent) }.onFailure {
                openAppNotificationSettings(context)
            }
        } else {
            openAppNotificationSettings(context)
        }
    }

    fun openAppNotificationSettings(context: Context) {
        val intent = Intent(Settings.ACTION_APP_NOTIFICATION_SETTINGS).apply {
            putExtra(Settings.EXTRA_APP_PACKAGE, context.packageName)
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        }
        runCatching { context.startActivity(intent) }
    }
}

object AndroidSystemNotificationDispatcher : SystemNotificationDispatcher {
    const val CHANNEL_STANDARD = "monitoring_standard"
    const val CHANNEL_ALARM = "monitoring_alarm"
    const val CHANNEL_SERVICE = "monitoring_service"
    const val CHANNEL_HOURLY = "ap_hourly_analysis"

    // Compatibility constants kept for existing callers/tests
    const val STRATEGY_CHANNEL_ID = CHANNEL_ALARM
    const val CHANNEL_CRITICAL_PATTERNS = CHANNEL_ALARM
    const val CHANNEL_GAME_DETECTION = CHANNEL_STANDARD
    const val CHANNEL_MONITORING_ERRORS = CHANNEL_SERVICE
    const val CHANNEL_HOURLY_ANALYSIS = CHANNEL_HOURLY

    fun resolveVersionedChannelId(baseId: String, settings: NotificationSettings): String {
        val v = settings.channelConfigVersion.coerceAtLeast(1)
        return if (v <= 1 || baseId == CHANNEL_SERVICE || baseId == CHANNEL_HOURLY) baseId else "${baseId}_v$v"
    }

    override fun hasNotificationPermission(context: Context): Boolean {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            ContextCompat.checkSelfPermission(
                context,
                Manifest.permission.POST_NOTIFICATIONS
            ) == PackageManager.PERMISSION_GRANTED
        } else {
            true
        }
    }

    override fun isDoNotDisturbBlocking(context: Context, priorityLevel: NotificationPriorityLevel): Boolean {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.M) return false
        val nm = context.getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager ?: return false
        return when (nm.currentInterruptionFilter) {
            NotificationManager.INTERRUPTION_FILTER_NONE -> true
            NotificationManager.INTERRUPTION_FILTER_ALARMS -> true
            NotificationManager.INTERRUPTION_FILTER_PRIORITY -> priorityLevel != NotificationPriorityLevel.URGENT
            else -> false
        }
    }

    fun resolveChannelId(
        type: StrategyNotificationType,
        settings: NotificationSettings = NotificationSettings()
    ): String {
        val base = when (type) {
            StrategyNotificationType.STRATEGY_REACHED,
            StrategyNotificationType.STRATEGY_APPROACHING -> CHANNEL_ALARM
            StrategyNotificationType.HOURLY_SUMMARY,
            StrategyNotificationType.NEW_HOUR_STARTED -> CHANNEL_HOURLY
            StrategyNotificationType.GAME_DETECTED,
            StrategyNotificationType.AUTOMATION_ACTION -> CHANNEL_STANDARD
            StrategyNotificationType.HISTORY_UNAVAILABLE,
            StrategyNotificationType.MONITORING_ERROR,
            StrategyNotificationType.STRATEGY_PAUSED,
            StrategyNotificationType.STRATEGY_INVALIDATED -> CHANNEL_SERVICE
        }
        return resolveVersionedChannelId(base, settings)
    }

    fun resolveTargetTab(type: StrategyNotificationType): Int = when (type) {
        StrategyNotificationType.HOURLY_SUMMARY,
        StrategyNotificationType.NEW_HOUR_STARTED -> MainActivity.TAB_HOURLY_ANALYSIS
        StrategyNotificationType.GAME_DETECTED,
        StrategyNotificationType.HISTORY_UNAVAILABLE,
        StrategyNotificationType.MONITORING_ERROR,
        StrategyNotificationType.AUTOMATION_ACTION -> MainActivity.TAB_LIVE_MONITOR
        StrategyNotificationType.STRATEGY_REACHED,
        StrategyNotificationType.STRATEGY_APPROACHING,
        StrategyNotificationType.STRATEGY_INVALIDATED,
        StrategyNotificationType.STRATEGY_PAUSED -> MainActivity.TAB_PATTERNS
    }

    fun ensureStrategyChannel(context: Context, settings: NotificationSettings) {
        ensureAllChannels(context, settings)
    }

    fun ensureAllChannels(context: Context, settings: NotificationSettings) {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.O) return
        val nm = context.getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager ?: return

        val alarmAttrs = AudioAttributes.Builder()
            .setUsage(AudioAttributes.USAGE_ALARM)
            .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
            .build()
        val alarmUri = AlarmAudioController.resolveSoundUri(settings.alarmSoundUri, isAlarm = true)

        val notifAttrs = AudioAttributes.Builder()
            .setUsage(AudioAttributes.USAGE_NOTIFICATION_EVENT)
            .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
            .build()
        val standardUri = AlarmAudioController.resolveSoundUri(settings.standardSoundUri, isAlarm = false)

        val standardChannelId = resolveVersionedChannelId(CHANNEL_STANDARD, settings)
        val alarmChannelId = resolveVersionedChannelId(CHANNEL_ALARM, settings)

        val standard = NotificationChannel(
            standardChannelId,
            "Standard Alerts",
            NotificationManager.IMPORTANCE_DEFAULT
        ).apply {
            description = "Standard notifications for game detection, monitoring updates, and resumption."
            val vibEnabled = settings.vibrationEnabled && settings.standardVibrationPattern != "NONE"
            enableVibration(vibEnabled)
            if (vibEnabled) {
                vibrationPattern = AlarmAudioController.vibrationTimingsFor(settings.standardVibrationPattern)
            }
            if (!settings.soundEnabled || !settings.standardAlertEnabled) {
                setSound(null, null)
            } else {
                setSound(standardUri, notifAttrs)
            }
        }

        val alarm = NotificationChannel(
            alarmChannelId,
            "Pattern Match Signal Alarm",
            NotificationManager.IMPORTANCE_HIGH
        ).apply {
            description = "Distinctive alarm notifications when a configured pattern or condition matches."
            val vibEnabled = settings.alarmVibrationPattern != "NONE"
            enableVibration(vibEnabled)
            if (vibEnabled) {
                vibrationPattern = AlarmAudioController.vibrationTimingsFor(settings.alarmVibrationPattern)
            }
            setSound(alarmUri, alarmAttrs)
            enableLights(true)
        }

        val service = NotificationChannel(
            CHANNEL_SERVICE,
            "Persistent Monitoring Service",
            NotificationManager.IMPORTANCE_LOW
        ).apply {
            description = "Persistent notification ensuring overlay and background monitoring remain active."
        }

        val hourly = NotificationChannel(
            CHANNEL_HOURLY,
            "Hourly Analysis",
            NotificationManager.IMPORTANCE_DEFAULT
        ).apply {
            description = "Hour transition summaries and statistical digests."
            enableVibration(false)
        }

        nm.createNotificationChannels(listOf(standard, alarm, service, hourly))
    }

    override fun postSystemNotification(
        context: Context,
        notificationId: Int,
        title: String,
        message: String,
        priorityLevel: NotificationPriorityLevel,
        soundEnabled: Boolean,
        vibrationEnabled: Boolean,
        historyRecordId: Long
    ): Boolean = postChannelNotification(
        context = context,
        channelId = CHANNEL_CRITICAL_PATTERNS,
        notificationId = notificationId,
        title = title,
        message = message,
        priorityLevel = priorityLevel,
        soundEnabled = soundEnabled,
        vibrationEnabled = vibrationEnabled,
        historyRecordId = historyRecordId
    )

    override fun postChannelNotification(
        context: Context,
        channelId: String,
        notificationId: Int,
        title: String,
        message: String,
        priorityLevel: NotificationPriorityLevel,
        soundEnabled: Boolean,
        vibrationEnabled: Boolean,
        historyRecordId: Long,
        targetTab: Int,
        targetGameId: String?
    ): Boolean {
        return try {
            val nm = context.getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager
                ?: return false
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O && nm.getNotificationChannel(channelId) == null) {
                ensureAllChannels(
                    context,
                    NotificationSettings(soundEnabled = soundEnabled, vibrationEnabled = vibrationEnabled)
                )
            }

            val openIntent = Intent(context, MainActivity::class.java).apply {
                flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_SINGLE_TOP
                putExtra(MainActivity.EXTRA_OPEN_TAB, targetTab)
                putExtra(MainActivity.EXTRA_NOTIFICATION_HISTORY_ID, historyRecordId)
                if (!targetGameId.isNullOrBlank()) {
                    putExtra(MainActivity.EXTRA_GAME_ID, targetGameId)
                }
            }
            val pendingIntent = PendingIntent.getActivity(
                context,
                notificationId,
                openIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
            )

            val compatPriority = when (priorityLevel) {
                NotificationPriorityLevel.URGENT -> NotificationCompat.PRIORITY_MAX
                NotificationPriorityLevel.HIGH -> NotificationCompat.PRIORITY_HIGH
                NotificationPriorityLevel.STANDARD -> NotificationCompat.PRIORITY_DEFAULT
            }

            val builder = NotificationCompat.Builder(context, channelId)
                .setSmallIcon(R.mipmap.ic_launcher_round)
                .setContentTitle(title)
                .setContentText(message)
                .setStyle(NotificationCompat.BigTextStyle().bigText(message))
                .setPriority(compatPriority)
                .setAutoCancel(true)
                .setContentIntent(pendingIntent)

            val isAlarmChannel = channelId.startsWith(CHANNEL_ALARM)
            if (!soundEnabled || !isAlarmChannel) {
                builder.setSilent(!soundEnabled)
            }
            if (vibrationEnabled && isAlarmChannel) {
                builder.setVibrate(longArrayOf(0L, 250L, 120L, 250L))
            }
            if (isAlarmChannel) {
                val stopIntent = Intent(context, AlarmStopReceiver::class.java).apply {
                    action = AlarmAudioController.ACTION_STOP_ALARM
                }
                val stopPi = PendingIntent.getBroadcast(
                    context,
                    2005,
                    stopIntent,
                    PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
                )
                builder.addAction(R.mipmap.ic_launcher_round, "STOP ALARM", stopPi)
            }

            nm.notify(notificationId, builder.build())
            true
        } catch (t: Throwable) {
            Log.e("StrategyNotifMgr", "Failed posting Android notification", t)
            false
        }
    }
}

/**
 * Central Android System Notification Center controller.
 * Routes alerts to Pattern Match Signal Alarm, Standard Alerts, Hourly Analysis, or Persistent Service;
 * respects Android 13+ POST_NOTIFICATIONS, Do Not Disturb, quiet hours, cooldowns, and deduplication.
 */
class StrategyNotificationManager(
    private val context: Context?,
    private val repository: ScreenEarnerRepository,
    private val dispatcher: SystemNotificationDispatcher = AndroidSystemNotificationDispatcher
) {
    private val lastDeliveredOccurrences = ConcurrentHashMap<String, Long>()
    private val lastStrategyDispatchTimes = ConcurrentHashMap<Long, Long>()
    private val strategyHourDispatchCounts = ConcurrentHashMap<Pair<Long, Long>, Int>()
    private val notificationIdSeq = AtomicInteger(2100)

    companion object {
        fun isWithinQuietHours(settings: NotificationSettings, nowMs: Long): Boolean {
            if (!settings.quietHoursEnabled) return false
            val cal = Calendar.getInstance().apply { timeInMillis = nowMs }
            val currentMinutes = cal.get(Calendar.HOUR_OF_DAY) * 60 + cal.get(Calendar.MINUTE)
            val startMinutes = settings.quietStartHour.coerceIn(0, 23) * 60 + settings.quietStartMinute.coerceIn(0, 59)
            val endMinutes = settings.quietEndHour.coerceIn(0, 23) * 60 + settings.quietEndMinute.coerceIn(0, 59)

            return if (startMinutes == endMinutes) {
                false
            } else if (startMinutes < endMinutes) {
                currentMinutes in startMinutes until endMinutes
            } else {
                currentMinutes >= startMinutes || currentMinutes < endMinutes
            }
        }
    }

    fun evaluateSuppressionReason(
        event: PatternAlertEvent,
        settings: NotificationSettings,
        nowMs: Long
    ): NotificationDeliveryStatus? {
        if (!settings.masterEnabled) {
            return NotificationDeliveryStatus.SUPPRESSED_MASTER_OFF
        }
        val isSignalAlarmEvent = event.notificationType == StrategyNotificationType.STRATEGY_REACHED ||
            event.notificationType == StrategyNotificationType.STRATEGY_APPROACHING
        if (isSignalAlarmEvent && !settings.highPriorityAlarmEnabled && !settings.standardAlertEnabled) {
            return NotificationDeliveryStatus.SUPPRESSED_TYPE_OFF
        }
        if (!isSignalAlarmEvent &&
            event.notificationType == StrategyNotificationType.GAME_DETECTED &&
            !settings.standardAlertEnabled
        ) {
            return NotificationDeliveryStatus.SUPPRESSED_TYPE_OFF
        }

        val typeEnabled = when (event.notificationType) {
            StrategyNotificationType.STRATEGY_APPROACHING -> settings.approachingEnabled
            StrategyNotificationType.STRATEGY_REACHED -> settings.reachedEnabled
            StrategyNotificationType.STRATEGY_INVALIDATED -> settings.invalidatedEnabled
            StrategyNotificationType.HISTORY_UNAVAILABLE ->
                settings.historyUnavailableEnabled && settings.failureAlertsEnabled
            StrategyNotificationType.MONITORING_ERROR ->
                (settings.monitoringErrorsEnabled || settings.historyUnavailableEnabled) && settings.failureAlertsEnabled
            StrategyNotificationType.STRATEGY_PAUSED -> settings.pausedEnabled
            StrategyNotificationType.HOURLY_SUMMARY,
            StrategyNotificationType.NEW_HOUR_STARTED -> settings.hourlySummaryEnabled
            StrategyNotificationType.GAME_DETECTED -> settings.gameDetectionEnabled
            StrategyNotificationType.AUTOMATION_ACTION ->
                settings.standardAlertEnabled && settings.executionAlertsEnabled
        }
        if (!typeEnabled) {
            return NotificationDeliveryStatus.SUPPRESSED_TYPE_OFF
        }
        if (event.strategy.id > 0L && !event.strategy.notificationEnabled) {
            return NotificationDeliveryStatus.SUPPRESSED_STRATEGY_OFF
        }
        if (isWithinQuietHours(settings, nowMs)) {
            return NotificationDeliveryStatus.SUPPRESSED_QUIET_HOURS
        }
        if (context != null && dispatcher.isDoNotDisturbBlocking(context, event.strategy.notificationLevel)) {
            return NotificationDeliveryStatus.SUPPRESSED_QUIET_HOURS
        }
        if (lastDeliveredOccurrences.containsKey(event.occurrenceKey)) {
            return NotificationDeliveryStatus.SUPPRESSED_DUPLICATE
        }
        val effectiveStrategyKey = if (event.strategy.id > 0L) {
            event.strategy.id
        } else {
            event.strategy.name.hashCode().toLong()
        }
        val effectiveCooldownMin = maxOf(
            if (isSignalAlarmEvent && settings.alarmRepeated) settings.alarmRepeatIntervalMinutes else settings.repeatIntervalMinutes,
            event.strategy.repeatCooldownMinutes
        ).coerceAtLeast(0)
        if (event.notificationType == StrategyNotificationType.STRATEGY_REACHED && effectiveCooldownMin > 0) {
            val lastTime = lastStrategyDispatchTimes[effectiveStrategyKey] ?: 0L
            if (lastTime > 0L && (nowMs - lastTime) < effectiveCooldownMin * 60_000L) {
                return NotificationDeliveryStatus.SUPPRESSED_COOLDOWN
            }
        }
        val hourBucket = nowMs / 3_600_000L
        val repeatCap = event.strategy.repeatLimit
        if (event.notificationType == StrategyNotificationType.STRATEGY_REACHED && repeatCap > 0) {
            val sentThisHour = strategyHourDispatchCounts[effectiveStrategyKey to hourBucket] ?: 0
            if (sentThisHour >= repeatCap) {
                return NotificationDeliveryStatus.SUPPRESSED_COOLDOWN
            }
        }
        return null
    }

    suspend fun dispatchPatternAlert(
        event: PatternAlertEvent,
        settings: NotificationSettings,
        nowMs: Long = System.currentTimeMillis(),
        gameId: String? = event.strategy.targetGameId.takeIf { it != "ALL" },
        occurrenceRecordId: Long? = null
    ): NotificationHistoryRecord {
        val channelId = AndroidSystemNotificationDispatcher.resolveChannelId(event.notificationType, settings)

        val suppression = evaluateSuppressionReason(event, settings, nowMs)
        if (suppression != null) {
            val suppressedRecord = NotificationHistoryRecord(
                strategyId = event.strategy.id,
                strategyName = event.strategy.name,
                gameId = gameId.orEmpty(),
                occurrenceRecordId = occurrenceRecordId,
                channelId = channelId,
                notificationType = event.notificationType,
                title = event.title,
                message = event.message,
                triggeringRecordKey = event.triggeringRecordKey,
                triggeringMultiplier = event.triggeringMultiplier,
                timestampMs = nowMs,
                deliveryStatus = suppression,
                interactionStatus = NotificationInteractionStatus.UNREAD
            )
            val savedId = repository.addNotificationHistory(suppressedRecord)
            return suppressedRecord.copy(id = savedId)
        }

        val appCtx = context
        val hasPermission = appCtx != null && dispatcher.hasNotificationPermission(appCtx)
        if (!hasPermission) {
            val deniedRecord = NotificationHistoryRecord(
                strategyId = event.strategy.id,
                strategyName = event.strategy.name,
                gameId = gameId.orEmpty(),
                occurrenceRecordId = occurrenceRecordId,
                channelId = channelId,
                notificationType = event.notificationType,
                title = event.title,
                message = event.message,
                triggeringRecordKey = event.triggeringRecordKey,
                triggeringMultiplier = event.triggeringMultiplier,
                timestampMs = nowMs,
                deliveryStatus = NotificationDeliveryStatus.PERMISSION_DENIED,
                interactionStatus = NotificationInteractionStatus.UNREAD
            )
            val savedId = repository.addNotificationHistory(deniedRecord)
            return deniedRecord.copy(id = savedId)
        }

        val pendingRecord = NotificationHistoryRecord(
            strategyId = event.strategy.id,
            strategyName = event.strategy.name,
            gameId = gameId.orEmpty(),
            occurrenceRecordId = occurrenceRecordId,
            channelId = channelId,
            notificationType = event.notificationType,
            title = event.title,
            message = event.message,
            triggeringRecordKey = event.triggeringRecordKey,
            triggeringMultiplier = event.triggeringMultiplier,
            timestampMs = nowMs,
            deliveryStatus = NotificationDeliveryStatus.DELIVERED,
            interactionStatus = NotificationInteractionStatus.UNREAD
        )
        val recordId = repository.addNotificationHistory(pendingRecord)

        val uniqueSeed = (event.occurrenceKey.hashCode() and 0x0FFFFFFF)
        val notifId = if (uniqueSeed > 0) uniqueSeed else notificationIdSeq.incrementAndGet()
        if (dispatcher === AndroidSystemNotificationDispatcher) {
            AndroidSystemNotificationDispatcher.ensureAllChannels(appCtx, settings)
        }
        val posted = dispatcher.postChannelNotification(
            context = appCtx,
            channelId = channelId,
            notificationId = notifId,
            title = event.title,
            message = event.message,
            priorityLevel = event.strategy.notificationLevel,
            soundEnabled = settings.soundEnabled,
            vibrationEnabled = settings.vibrationEnabled,
            historyRecordId = recordId,
            targetTab = AndroidSystemNotificationDispatcher.resolveTargetTab(event.notificationType),
            targetGameId = gameId
        )

        if (posted &&
            event.notificationType == StrategyNotificationType.STRATEGY_REACHED &&
            event.strategy.useAlarmStyle &&
            settings.highPriorityAlarmEnabled &&
            dispatcher === AndroidSystemNotificationDispatcher
        ) {
            val activeProfile = runCatching {
                repository.getAlarmProfile(event.strategy.resolveAlarmProfileCategory())
            }.getOrNull()
            if (activeProfile == null || activeProfile.enabled) {
                val effectiveSettings = if (activeProfile != null) {
                    settings.copy(
                        alarmSoundUri = activeProfile.soundUri.ifBlank { settings.alarmSoundUri },
                        alarmVibrationPattern = if (event.strategy.vibrationEnabled && activeProfile.vibrationEnabled) {
                            activeProfile.vibrationPattern
                        } else {
                            "NONE"
                        },
                        alarmRepeated = activeProfile.repeatEnabled && settings.alarmRepeated
                    )
                } else {
                    settings.copy(
                        alarmVibrationPattern = if (event.strategy.vibrationEnabled) settings.alarmVibrationPattern else "NONE"
                    )
                }
                AlarmAudioController.startAlarm(
                    context = appCtx,
                    settings = effectiveSettings,
                    title = "SIGNAL DETECTED: ${event.strategy.name}",
                    nowMs = nowMs
                )
            }
        }

        val effectiveStrategyKey = if (event.strategy.id > 0L) {
            event.strategy.id
        } else {
            event.strategy.name.hashCode().toLong()
        }
        val finalStatus = if (posted) {
            lastDeliveredOccurrences[event.occurrenceKey] = nowMs
            if (event.notificationType == StrategyNotificationType.STRATEGY_REACHED) {
                lastStrategyDispatchTimes[effectiveStrategyKey] = nowMs
                val hourBucket = nowMs / 3_600_000L
                val key = effectiveStrategyKey to hourBucket
                strategyHourDispatchCounts[key] = (strategyHourDispatchCounts[key] ?: 0) + 1
            }
            NotificationDeliveryStatus.DELIVERED
        } else {
            NotificationDeliveryStatus.FAILED
        }

        val finalRecord = pendingRecord.copy(id = recordId, deliveryStatus = finalStatus)
        if (!posted) {
            repository.addNotificationHistory(finalRecord)
        }
        return finalRecord
    }

    suspend fun dispatchSystemNotice(
        strategyName: String = "Predict Manager Monitor",
        type: StrategyNotificationType,
        title: String,
        message: String,
        occurrenceKey: String,
        settings: NotificationSettings,
        nowMs: Long = System.currentTimeMillis(),
        gameId: String? = null
    ): NotificationHistoryRecord {
        val dummyStrategy = CustomPatternStrategy(id = 0L, name = strategyName)
        val event = PatternAlertEvent(
            strategy = dummyStrategy,
            notificationType = type,
            title = title,
            message = message,
            occurrenceKey = occurrenceKey,
            timestampMs = nowMs
        )
        return dispatchPatternAlert(event, settings, nowMs, gameId = gameId)
    }

    suspend fun dispatchHourlyBoundaryNotice(
        gameId: String,
        gameName: String,
        closedHourLabel: String,
        newHourLabel: String,
        summaryLine: String,
        settings: NotificationSettings,
        nowMs: Long = System.currentTimeMillis()
    ): NotificationHistoryRecord = dispatchSystemNotice(
        strategyName = "$gameName Hourly Digest",
        type = StrategyNotificationType.NEW_HOUR_STARTED,
        title = "New Hour $newHourLabel — $closedHourLabel Finalized",
        message = summaryLine,
        occurrenceKey = "hour_${gameId}_${newHourLabel}_${nowMs / 3_600_000L}",
        settings = settings,
        nowMs = nowMs,
        gameId = gameId
    )

    fun clearSessionCache() {
        lastDeliveredOccurrences.clear()
        lastStrategyDispatchTimes.clear()
        strategyHourDispatchCounts.clear()
    }
}
