# AutoTouch – Android typing & touch automation

Runs a simple macro script (tap, swipe, long-press, type, click-by-text…) inside any app,
controlled from a floating panel. Built on Android's Accessibility Service, which is the
only supported way for an app to touch and type into other apps without root.

## Build & install
1. Open the `AutoTouch` folder in Android Studio (Koala or newer). Let Gradle sync –
   it downloads Gradle 8.7, AGP 8.5 and Kotlin 1.9 automatically.
2. Plug in your phone (USB debugging on) and press ▶ Run. Min Android version: 8.0.

## First-time setup on the phone
1. Open AutoTouch → **Open Accessibility settings** → Installed apps / Downloaded apps →
   **AutoTouch automation** → turn ON.
   - Android 13+ shows "Restricted setting" for sideloaded apps. Fix: Settings ▸ Apps ▸
     AutoTouch ▸ ⋮ menu (top right) ▸ **Allow restricted settings**, then try again.
2. Back in AutoTouch, write your script → **Save & show floating panel**.
3. Open the target app, press **▶ Run** on the panel. Drag the panel by its title.

## Panel buttons
- **▶ Run / ■ Stop** – run the saved script (loops and step delay come from the main screen)
- **📍 Pick** – tap anywhere; adds `tap X Y` to the end of your script
- **✕** – hide the panel (also stops a running macro)

## Commands
See the help text at the bottom of the main screen, or `MacroParser.kt`.

## Notes
- Prefer `click TEXT` / `focus TEXT` over coordinates – they survive layout changes.
  `click` waits up to 3 s for the element to appear.
- `type` sets the field's text directly (instant, works with any keyboard). Use
  `typeslow` for apps that check each keystroke (search-as-you-type boxes).
- Keep the panel away from spots your script taps – a tap on the panel hits the panel.
- Games and some banking apps block accessibility input or draw without views;
  coordinates still work in most games, text-based commands won't.

## Automatic APK builds (GitHub Actions)
Every push to `main` builds the app on GitHub's servers. Get the APK from
**Releases ▸ AutoTouch (latest build) ▸ AutoTouch.apk** – you can open that link
straight from your phone. You can also start a build by hand from **Actions ▸ Build APK ▸ Run workflow**.

/**
 * The engine. Android only lets an app touch/type into OTHER apps through an
 * Accessibility Service, which the user must switch on in Settings.
 */
class AutoService : AccessibilityService() {

    companion object {
        /** Non-null while the service is enabled and connected. */
        var instance: AutoService? = null
            private set
    }

    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Main)
    private var job: Job? = null
    private lateinit var overlay: OverlayController

    val isRunning: Boolean get() = job?.isActive == true

    override fun onServiceConnected() {
        super.onServiceConnected()
        instance = this
        overlay = OverlayController(this)
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) { /* not needed */ }

    override fun onInterrupt() = stopMacro()

    override fun onUnbind(intent: Intent?): Boolean {
        cleanup()
        return super.onUnbind(intent)
    }

    override fun onDestroy() {
        cleanup()
        super.onDestroy()
    }

    private fun cleanup() {
        stopMacro()
        if (::overlay.isInitialized) overlay.hide()
        instance = null
    }

    // ---------------------------------------------------------------- panel

    fun showPanel() = overlay.show()
    fun hidePanel() = overlay.hide()

    // ---------------------------------------------------------------- running

    /** loops <= 0 means repeat until Stop is pressed. */
    fun runMacro(script: String, loops: Int, stepDelay: Long) {
        if (isRunning) return
        val steps = try {
            MacroParser.parse(script)
        } catch (e: MacroParser.ParseError) {
            overlay.setStatus("⚠ ${e.message}")
            return
        }
        overlay.setRunning(true)
        job = scope.launch {
            try {
                val total = if (loops <= 0) Int.MAX_VALUE else loops
                var failures = 0
                for (loop in 1..total) {
                    for (step in steps) {
                        ensureActive()
                        val loopLabel = if (loops <= 0) "Loop $loop" else "Loop $loop/$loops"
                        overlay.setStatus("$loopLabel · ${step.label}")
                        if (!execute(step)) {
                            failures++
                            overlay.setStatus("✗ Failed: ${step.label}")
                            delay(600)
                        }
                        delay(stepDelay)
                    }
                }
                overlay.setStatus(if (failures == 0) "✓ Done" else "Done with $failures failed step(s)")
            } catch (e: CancellationException) {
                overlay.setStatus("■ Stopped")
                throw e
            } finally {
                overlay.setRunning(false)
            }
        }
    }

    fun stopMacro() {
        job?.cancel()
        job = null
    }

    private suspend fun execute(step: Step): Boolean = when (step) {
        is Step.Tap -> tap(step.x, step.y)
        is Step.DoubleTap -> tap(step.x, step.y).also { delay(90) } && tap(step.x, step.y)
        is Step.LongPress -> gesture(pointPath(step.x, step.y), step.ms)
        is Step.Swipe -> gesture(Path().apply {
            moveTo(step.x1, step.y1); lineTo(step.x2, step.y2)
        }, step.ms)
        is Step.Wait -> { delay(step.ms); true }
        is Step.Type -> setText(step.text, append = false)
        is Step.Append -> setText(step.text, append = true)
        is Step.TypeSlow -> typeSlow(step.text, step.charDelay)
        is Step.Clear -> setText("", append = false)
        is Step.Click -> clickElement(step.query)
        is Step.Focus -> focusField(step.query)
        is Step.Enter -> pressEnter()
        is Step.Global -> performGlobalAction(step.action)
    }

    // ---------------------------------------------------------------- touch

    private fun pointPath(x: Float, y: Float) = Path().apply { moveTo(x, y) }

    suspend fun tap(x: Float, y: Float) = gesture(pointPath(x, y), 50)

    private suspend fun gesture(path: Path, durationMs: Long): Boolean =
        suspendCancellableCoroutine { cont ->
            val duration = durationMs.coerceIn(1, GestureDescription.getMaxGestureDuration())
            val gesture = GestureDescription.Builder()
                .addStroke(GestureDescription.StrokeDescription(path, 0, duration))
                .build()
            val accepted = dispatchGesture(gesture, object : GestureResultCallback() {
                override fun onCompleted(g: GestureDescription?) { if (cont.isActive) cont.resume(true) }
                override fun onCancelled(g: GestureDescription?) { if (cont.isActive) cont.resume(false) }
            }, null)
            if (!accepted && cont.isActive) cont.resume(false)
        }

    // ---------------------------------------------------------------- typing

    private fun focusedInput(): AccessibilityNodeInfo? =
        rootInActiveWindow?.findFocus(AccessibilityNodeInfo.FOCUS_INPUT)

    private fun setText(text: String, append: Boolean, node: AccessibilityNodeInfo? = focusedInput()): Boolean {
        val target = node ?: return false
        val existing = if (append && !target.isShowingHintText) target.text?.toString().orEmpty() else ""
        val args = Bundle().apply {
            putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, existing + text)
        }
        return target.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args)
    }

    /** Types character by character so apps that react to each keystroke still work. */
    private suspend fun typeSlow(text: String, charDelay: Long): Boolean {
        val node = focusedInput() ?: return false
        val start = if (node.isShowingHintText) "" else node.text?.toString().orEmpty()
        val sb = StringBuilder(start)
        for (c in text) {
            sb.append(c)
            node.refresh()
            val args = Bundle().apply {
                putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, sb.toString())
            }
            if (!node.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args)) return false
            delay(charDelay)
        }
        return true
    }

    private fun pressEnter(): Boolean {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.R) return false
        val node = focusedInput() ?: return false
        return node.performAction(AccessibilityNodeInfo.AccessibilityAction.ACTION_IME_ENTER.id)
    }

    // ---------------------------------------------------------------- finding elements

    /** Waits up to [timeoutMs] for an element to appear on screen. */
    private suspend fun waitForNode(query: String, editableOnly: Boolean, timeoutMs: Long = 3000): AccessibilityNodeInfo? {
        val until = System.currentTimeMillis() + timeoutMs
        while (true) {
            findNode(query, editableOnly)?.let { return it }
            if (System.currentTimeMillis() > until) return null
            delay(250)
        }
    }

    private fun findNode(query: String, editableOnly: Boolean): AccessibilityNodeInfo? {
        val q = query.trim().lowercase()
        val roots = windows.mapNotNull { it.root }.ifEmpty { listOfNotNull(rootInActiveWindow) }
            .filter { it.packageName != packageName } // never match our own panel

        var partial: AccessibilityNodeInfo? = null
        for (root in roots) {
            val queue = ArrayDeque<AccessibilityNodeInfo>().apply { add(root) }
            while (queue.isNotEmpty()) {
                val n = queue.removeFirst()
                for (i in 0 until n.childCount) n.getChild(i)?.let(queue::add)
                if (editableOnly && !n.isEditable) continue
                if (!n.isVisibleToUser) continue

                val candidates = listOfNotNull(
                    n.text?.toString(),
                    n.contentDescription?.toString(),
                    n.hintText?.toString(),
                    n.viewIdResourceName?.substringAfter(":id/"),
                ).map { it.trim().lowercase() }

                if (candidates.any { it == q }) return n          // exact match wins
                if (partial == null && candidates.any { it.contains(q) }) partial = n
            }
        }
        return partial
    }

    private suspend fun clickElement(query: String): Boolean {
        val node = waitForNode(query, editableOnly = false) ?: return false
        // Text labels are often inside a clickable parent (e.g. a button row).
        var target: AccessibilityNodeInfo? = node
        while (target != null && !target.isClickable) target = target.parent
        if (target != null && target.performAction(AccessibilityNodeInfo.ACTION_CLICK)) return true
        // Fallback: physically tap the centre of the element.
        val r = Rect().also(node::getBoundsInScreen)
        return tap(r.exactCenterX(), r.exactCenterY())
    }

    private suspend fun focusField(query: String): Boolean {
        val node = waitForNode(query, editableOnly = true) ?: return false
        if (node.performAction(AccessibilityNodeInfo.ACTION_FOCUS)) {
            node.performAction(AccessibilityNodeInfo.ACTION_CLICK) // brings up the keyboard in most apps
            return true
        }
        val r = Rect().also(node::getBoundsInScreen)
        return tap(r.exactCenterX(), r.exactCenterY())
    }
}

package com.autotouch.app

import android.accessibilityservice.AccessibilityService

/** One action in a macro script. */
sealed class Step(val label: String) {
    class Tap(val x: Float, val y: Float) : Step("tap ${x.toInt()},${y.toInt()}")
    class DoubleTap(val x: Float, val y: Float) : Step("doubletap ${x.toInt()},${y.toInt()}")
    class LongPress(val x: Float, val y: Float, val ms: Long) : Step("longpress ${x.toInt()},${y.toInt()}")
    class Swipe(val x1: Float, val y1: Float, val x2: Float, val y2: Float, val ms: Long) :
        Step("swipe ${x1.toInt()},${y1.toInt()} → ${x2.toInt()},${y2.toInt()}")
    class Wait(val ms: Long) : Step("wait ${ms}ms")
    class Type(val text: String) : Step("type \"${text.take(20)}\"")
    class Append(val text: String) : Step("append \"${text.take(20)}\"")
    class TypeSlow(val charDelay: Long, val text: String) : Step("typeslow \"${text.take(20)}\"")
    class Clear : Step("clear")
    class Click(val query: String) : Step("click \"$query\"")
    class Focus(val query: String) : Step("focus \"$query\"")
    class Enter : Step("enter")
    class Global(val action: Int, name: String) : Step(name)
}

/**
 * Parses a plain-text macro script. One command per line; lines starting with # are comments.
 *
 *   tap X Y                     single tap at screen coordinates
 *   doubletap X Y
 *   longpress X Y [ms]          default 800 ms
 *   swipe X1 Y1 X2 Y2 [ms]      default 300 ms
 *   wait MS
 *   type TEXT                   replace text in the focused input field
 *   append TEXT                 add to the end of the focused field
 *   typeslow MS TEXT            type one character at a time, MS between characters
 *   clear                       empty the focused field
 *   click TEXT                  click the element whose text/description/hint/id matches
 *   focus TEXT                  focus the input field matching TEXT (text, hint or id)
 *   enter                       press the keyboard's Enter/Send/Search action (Android 11+)
 *   back | home | recents | notifications
 */
object MacroParser {
    class ParseError(msg: String) : Exception(msg)

    fun parse(script: String): List<Step> {
        val steps = mutableListOf<Step>()
        script.lines().forEachIndexed { index, raw ->
            val line = raw.trim()
            if (line.isEmpty() || line.startsWith("#")) return@forEachIndexed
            val lineNo = index + 1
            val cmd = line.substringBefore(' ').lowercase()
            val rest = line.substringAfter(' ', "").trim()

            fun nums(min: Int): List<Float> {
                val parts = rest.split(Regex("\\s+")).filter { it.isNotEmpty() }
                val values = parts.map {
                    it.toFloatOrNull() ?: throw ParseError("Line $lineNo: '$it' is not a number")
                }
                if (values.size < min) throw ParseError("Line $lineNo: '$cmd' needs at least $min numbers")
                values.forEach { if (it < 0) throw ParseError("Line $lineNo: coordinates can't be negative") }
                return values
            }

            fun text(): String {
                if (rest.isEmpty()) throw ParseError("Line $lineNo: '$cmd' needs some text")
                return unquote(rest)
            }

            steps += when (cmd) {
                "tap" -> nums(2).let { Step.Tap(it[0], it[1]) }
                "doubletap" -> nums(2).let { Step.DoubleTap(it[0], it[1]) }
                "longpress" -> nums(2).let { Step.LongPress(it[0], it[1], it.getOrNull(2)?.toLong() ?: 800L) }
                "swipe" -> nums(4).let { Step.Swipe(it[0], it[1], it[2], it[3], it.getOrNull(4)?.toLong() ?: 300L) }
                "wait", "sleep", "delay" -> nums(1).let { Step.Wait(it[0].toLong()) }
                "type" -> Step.Type(text())
                "append" -> Step.Append(text())
                "typeslow" -> {
                    val delayPart = rest.substringBefore(' ')
                    val ms = delayPart.toLongOrNull()
                        ?: throw ParseError("Line $lineNo: use 'typeslow MS TEXT', e.g. typeslow 80 Hello")
                    val t = rest.substringAfter(' ', "").trim()
                    if (t.isEmpty()) throw ParseError("Line $lineNo: 'typeslow' needs some text")
                    Step.TypeSlow(ms, unquote(t))
                }
                "clear" -> Step.Clear()
                "click" -> Step.Click(text())
                "focus" -> Step.Focus(text())
                "enter" -> Step.Enter()
                "back" -> Step.Global(AccessibilityService.GLOBAL_ACTION_BACK, "back")
                "home" -> Step.Global(AccessibilityService.GLOBAL_ACTION_HOME, "home")
                "recents" -> Step.Global(AccessibilityService.GLOBAL_ACTION_RECENTS, "recents")
                "notifications" -> Step.Global(AccessibilityService.GLOBAL_ACTION_NOTIFICATIONS, "notifications")
                else -> throw ParseError("Line $lineNo: unknown command '$cmd'")
            }
        }
        if (steps.isEmpty()) throw ParseError("Script has no commands")
        return steps
    }

    private fun unquote(s: String): String =
        if (s.length >= 2 && s.first() == '"' && s.last() == '"') s.substring(1, s.length - 1) else s
}

        findViewById<TextView>(R.id.helpText).text = HELP

        findViewById<Button>(R.id.btnEnable).setOnClickListener {
            startActivity(Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS))
        }

        findViewById<Button>(R.id.btnSave).setOnClickListener {
            if (save()) Toast.makeText(this, "Saved", Toast.LENGTH_SHORT).show()
        }

        findViewById<Button>(R.id.btnPanel).setOnClickListener {
            if (!save()) return@setOnClickListener
            val service = AutoService.instance
            if (service == null) {
                Toast.makeText(this, "Turn on the AutoTouch accessibility service first", Toast.LENGTH_LONG).show()
                startActivity(Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS))
            } else {
                service.showPanel()
                Toast.makeText(this, "Panel shown – open the app you want to automate", Toast.LENGTH_LONG).show()
            }
        }
    }

    override fun onResume() {
        super.onResume()
        // Reload in case the 📍 Pick button appended new lines while we were away.
        scriptInput.setText(ScriptStore.script(this))
        loopsInput.setText(ScriptStore.loops(this).toString())
        delayInput.setText(ScriptStore.stepDelay(this).toString())
        updateServiceStatus()
    }

    override fun onPause() {
        super.onPause()
        save(quiet = true)
    }

    private fun updateServiceStatus() {
        val on = AutoService.instance != null
        serviceStatus.text = if (on) "● Accessibility service is ON" else "○ Accessibility service is OFF – tap the button below and enable \"AutoTouch automation\""
        serviceStatus.setBackgroundColor(if (on) 0x331E8E3E else 0x33D93025)
        serviceStatus.setTextColor(if (on) Color.rgb(30, 142, 62) else Color.rgb(217, 48, 37))
    }

    /** Validates and saves. Returns false (and shows the error) if the script is invalid. */
    private fun save(quiet: Boolean = false): Boolean {
        val script = scriptInput.text.toString()
        val loops = loopsInput.text.toString().toIntOrNull()?.coerceAtLeast(0) ?: 1
        val delay = delayInput.text.toString().toLongOrNull()?.coerceAtLeast(0) ?: 300L
        if (!quiet) {
            try {
                MacroParser.parse(script)
            } catch (e: MacroParser.ParseError) {
                Toast.makeText(this, e.message, Toast.LENGTH_LONG).show()
                return false
            }
        }
        ScriptStore.save(this, script, loops, delay)
        return true
    }

    private companion object {
        val HELP = """
            COMMANDS (one per line, # = comment)

/**
 * Small draggable control panel that floats over every app.
 * Uses TYPE_ACCESSIBILITY_OVERLAY, so no extra "draw over other apps" permission is needed.
 */
class OverlayController(private val service: AutoService) {

    private val wm = service.getSystemService(Context.WINDOW_SERVICE) as WindowManager
    private val density = service.resources.displayMetrics.density
    private fun dp(v: Int) = (v * density).toInt()

    private var panel: View? = null
    private var pickView: View? = null
    private var statusView: TextView? = null
    private var runButton: TextView? = null

    private val panelParams = WindowManager.LayoutParams(
        WindowManager.LayoutParams.WRAP_CONTENT,

            setPadding(0, dp(2), 0, dp(2))
        }

        val status = TextView(ctx).apply {
            text = "Ready"
            setTextColor(0xFFBDC1C6.toInt())
            textSize = 12f
            maxWidth = dp(230)
            maxLines = 2
        }

        val row = LinearLayout(ctx).apply { orientation = LinearLayout.HORIZONTAL }

        fun button(label: String, color: Int, onClick: () -> Unit) = TextView(ctx).apply {
            text = label
            setTextColor(Color.WHITE)
            textSize = 13f
            gravity = Gravity.CENTER
            setPadding(dp(12), dp(8), dp(12), dp(8))
            background = GradientDrawable().apply {
                setColor(color)
                cornerRadius = dp(8).toFloat()
            }
            setOnClickListener { onClick() }
            layoutParams = LinearLayout.LayoutParams(
                LinearLayout.LayoutParams.WRAP_CONTENT,
                LinearLayout.LayoutParams.WRAP_CONTENT
            ).apply { marginEnd = dp(6); topMargin = dp(8) }
        }

        val run = button("▶ Run", GREEN) { toggleRun() }
        row.addView(run)
        row.addView(button("📍 Pick", BLUE) { startPick() })
        row.addView(button("✕", GREY) { hide() })

        root.addView(title)
        root.addView(status)
        root.addView(row)

        // Drag the panel by its title bar.
        var startX = 0; var startY = 0; var touchX = 0f; var touchY = 0f
        title.setOnTouchListener { _, e ->
            when (e.action) {
                MotionEvent.ACTION_DOWN -> {
                    startX = panelParams.x; startY = panelParams.y
                    touchX = e.rawX; touchY = e.rawY
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    panelParams.x = startX + (e.rawX - touchX).toInt()
                    panelParams.y = startY + (e.rawY - touchY).toInt()
                    wm.updateViewLayout(root, panelParams)
                    true
                }
                else -> false
            }
        }

        wm.addView(root, panelParams)
        panel = root
        statusView = status
        runButton = run
        setRunning(service.isRunning)
    }

    fun hide() {
        service.stopMacro()
        removePick()
        panel?.let { runCatching { wm.removeView(it) } }
        panel = null
        statusView = null
        runButton = null
    }

    fun setStatus(text: String) {
        statusView?.text = text
    }

    fun setRunning(running: Boolean) {
        runButton?.apply {
            text = if (running) "■ Stop" else "▶ Run"
            (background as? GradientDrawable)?.setColor(if (running) RED else GREEN)
        }
    }

    private fun toggleRun() {
        if (service.isRunning) {
            service.stopMacro()
        } else {
            removePick()
            service.runMacro(
                ScriptStore.script(service),
                ScriptStore.loops(service),
                ScriptStore.stepDelay(service)
            )
        }
    }

    /** Full-screen catcher: the next touch anywhere becomes a "tap X Y" line in the script. */
    @SuppressLint("ClickableViewAccessibility", "SetTextI18n")
    private fun startPick() {
        if (pickView != null || service.isRunning) return
        val catcher = TextView(service).apply {
            text = "Tap anywhere to capture its coordinates"
            setTextColor(Color.WHITE)
            textSize = 16f
            gravity = Gravity.CENTER
            setBackgroundColor(0x441A73E8)
            setOnTouchListener { _, e ->
                if (e.action == MotionEvent.ACTION_DOWN) {
                    val x = e.rawX.toInt()
                    val y = e.rawY.toInt()
                    ScriptStore.appendLine(service, "tap $x $y")
                    setStatus("Added: tap $x $y")
                    Toast.makeText(service, "Added: tap $x $y", Toast.LENGTH_SHORT).show()
                    removePick()
                }
                true
            }
        }
        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.MATCH_PARENT,
            WindowManager.LayoutParams.MATCH_PARENT,
            WindowManager.LayoutParams.TYPE_ACCESSIBILITY_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
            PixelFormat.TRANSLUCENT
        ).apply { gravity = Gravity.TOP or Gravity.START }
        wm.addView(catcher, params)
        pickView = catcher
        setStatus("Pick mode: tap a spot…")
    }

    private fun removePick() {
        pickView?.let { runCatching { wm.removeView(it) } }
        pickView = null
    }

    private companion object {
        const val GREEN = 0xFF1E8E3E.toInt()
        const val BLUE = 0xFF1A73E8.toInt()
        const val RED = 0xFFD93025.toInt()
        const val GREY = 0xFF5F6368.toInt()
    }
}

package com.autotouch.app

import android.content.Context

/** Saves the script and run settings so the app and the floating panel share them. */
object ScriptStore {
    private const val PREFS = "autotouch"

    val DEFAULT_SCRIPT = """
        # AutoTouch script – one command per line
        # Use the 📍 Pick button on the floating panel to grab tap coordinates.
        wait 1000
        tap 540 1200
        type Hello from AutoTouch!
        enter
    """.trimIndent()

    private fun prefs(ctx: Context) = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)

    fun script(ctx: Context): String = prefs(ctx).getString("script", null) ?: DEFAULT_SCRIPT
    fun loops(ctx: Context): Int = prefs(ctx).getInt("loops", 1)
    fun stepDelay(ctx: Context): Long = prefs(ctx).getLong("stepDelay", 300L)

    fun save(ctx: Context, script: String, loops: Int, stepDelay: Long) {
        prefs(ctx).edit()
            .putString("script", script)
            .putInt("loops", loops)
            .putLong("stepDelay", stepDelay)
            .apply()
    }

    fun appendLine(ctx: Context, line: String) {
        val current = script(ctx).trimEnd()
        prefs(ctx).edit().putString("script", "$current\n$line").apply()
    }
}
